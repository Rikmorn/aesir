**Migration pattern (same as 12-04):**

BEFORE:
```typescript
import { logger } from "@aesir/common";
export const agentLogger = logger.child({ agentId: "dev-agent" });
agentLogger.info("action_name", { outcome: "success", message: "...", data: value });
```

AFTER:
```typescript
import { createLogger, createChildLogger } from "@aesir/common";
const logger = createLogger({ component: "agents:dev-agent" });
// or for child loggers with correlation:
const taskLogger = createChildLogger(logger, { correlationId: "agent_xxx", taskId });
logger.info({ data: value }, "Action completed successfully");
```
