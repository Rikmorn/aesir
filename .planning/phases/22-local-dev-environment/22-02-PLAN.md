  <action>
Add three new services to docker-compose.yml after the postgresql service and before the dev-agent service:

```yaml
  # Linear Integration - handles Linear webhooks and MCP tools
  linear-integration:
    build:
      context: .
      dockerfile: packages/integrations/linear/Dockerfile
    container_name: aesir-linear-integration
    depends_on:
      postgresql:
        condition: service_healthy
        restart: true
    environment:
      # Database connection
      - DATABASE_HOST=postgresql
      - DATABASE_PORT=5432
      - DATABASE_USER=temporal
      - DATABASE_PASSWORD=temporal
      - DATABASE_NAME=temporal
      - CREDENTIAL_ENCRYPTION_KEY=${CREDENTIAL_ENCRYPTION_KEY}
      # Linear configuration
      - LINEAR_WEBHOOK_SECRET=${LINEAR_WEBHOOK_SECRET}
      - LINEAR_CLIENT_ID=${LINEAR_CLIENT_ID}
      - LINEAR_CLIENT_SECRET=${LINEAR_CLIENT_SECRET}
      # Server configuration
      - PORT=3001
      - NODE_ENV=${NODE_ENV:-production}
    ports:
      - "3001:3001"
    healthcheck:
      test: ["CMD-SHELL", "node -e \"require('http').get('http://localhost:3001/health', (r) => process.exit(r.statusCode === 200 ? 0 : 1))\""]
      interval: 10s
      timeout: 5s
      retries: 3
      start_period: 30s
    networks:
      - aesir-network
    restart: unless-stopped

  # GitHub Integration - handles GitHub webhooks and MCP tools
  github-integration:
    build:
      context: .
      dockerfile: packages/integrations/github/Dockerfile
    container_name: aesir-github-integration
    depends_on:
      postgresql:
        condition: service_healthy
        restart: true
    environment:
      # Database connection
      - DATABASE_HOST=postgresql
      - DATABASE_PORT=5432
      - DATABASE_USER=temporal
      - DATABASE_PASSWORD=temporal
      - DATABASE_NAME=temporal
      - CREDENTIAL_ENCRYPTION_KEY=${CREDENTIAL_ENCRYPTION_KEY}
      # GitHub configuration
      - GITHUB_WEBHOOK_SECRET=${GITHUB_WEBHOOK_SECRET}
      - GITHUB_CLIENT_ID=${GITHUB_CLIENT_ID}
      - GITHUB_CLIENT_SECRET=${GITHUB_CLIENT_SECRET}
      # Server configuration
      - PORT=3002
      - NODE_ENV=${NODE_ENV:-production}
    ports:
      - "3002:3002"
    healthcheck:
      test: ["CMD-SHELL", "node -e \"require('http').get('http://localhost:3002/health', (r) => process.exit(r.statusCode === 200 ? 0 : 1))\""]
      interval: 10s
      timeout: 5s
      retries: 3
      start_period: 30s
    networks:
      - aesir-network
    restart: unless-stopped

  # Slack Integration - handles Slack events and MCP tools
  slack-integration:
    build:
      context: .
      dockerfile: packages/integrations/slack/Dockerfile
    container_name: aesir-slack-integration
    depends_on:
      postgresql:
        condition: service_healthy
        restart: true
    environment:
      # Database connection
      - DATABASE_HOST=postgresql
      - DATABASE_PORT=5432
      - DATABASE_USER=temporal
      - DATABASE_PASSWORD=temporal
      - DATABASE_NAME=temporal
      - CREDENTIAL_ENCRYPTION_KEY=${CREDENTIAL_ENCRYPTION_KEY}
      # Slack configuration
      - SLACK_SIGNING_SECRET=${SLACK_SIGNING_SECRET}
      - SLACK_CLIENT_ID=${SLACK_CLIENT_ID}
      - SLACK_CLIENT_SECRET=${SLACK_CLIENT_SECRET}
      - SLACK_BOT_TOKEN=${SLACK_BOT_TOKEN}
      - SLACK_MODE=http
      # Server configuration
      - PORT=3003
      - NODE_ENV=${NODE_ENV:-production}
    ports:
      - "3003:3003"
    healthcheck:
      test: ["CMD-SHELL", "node -e \"require('http').get('http://localhost:3003/health', (r) => process.exit(r.statusCode === 200 ? 0 : 1))\""]
      interval: 10s
      timeout: 5s
      retries: 3
      start_period: 30s
    networks:
      - aesir-network
    restart: unless-stopped
```
