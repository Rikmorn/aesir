```typescript
function createPickupTaskNode(linearClient: LinearClient) {
  return async function pickupTaskNode(state: DevWorkflowState): Promise<Partial<DevWorkflowState>> {
    // Read task from Linear
    const issue = await readIssue(linearClient, state.taskId);

    // Update status to "In Progress"
    await updateIssueStatus(linearClient, state.taskId, "started");

    // Emit thought activity
    await emitThought(linearClient, state.taskId, `Starting work on: ${issue.title}`);

    return {
      taskDescription: `${issue.title}\n\n${issue.description || ""}`,
      status: "coding"
    };
  };
}
```

**commitPRNode (factory pattern):**
```typescript
function createCommitPRNode(
  octokit: Octokit,
  linearClient: LinearClient,
  config: { owner: string; repo: string }
) {
  return async function commitPRNode(state: DevWorkflowState): Promise<Partial<DevWorkflowState>> {
    // Convert FileChange[] to format for createCommit
    const files = state.files
      .filter(f => f.operation !== "delete")
      .map(f => ({ path: f.path, content: f.content }));

    // Create commit
    await createCommit(octokit, {
      owner: config.owner,
      repo: config.repo,
      branch: state.branchName!,
      message: `feat: implement ${state.taskId}`,
      files
    });

    // Create PR
    const pr = await createPullRequest(octokit, {
      owner: config.owner,
      repo: config.repo,
      title: `[${state.taskId}] ${state.taskDescription.split('\n')[0]}`,
      body: `Implements ${state.taskId}\n\nGenerated by Dev Agent.`,
      head: state.branchName!,
      base: "main"
    });

    // Update Linear status to Done
    await updateIssueStatus(linearClient, state.taskId, "completed");

    // Emit response with PR link
    await emitResponse(linearClient, state.taskId, `Opened PR: ${pr.html_url}`);

    return {
      prNumber: pr.number,
      status: "complete"
    };
  };
}
```

```typescript
async function runDevWorkflow(
  taskId: string,
  deps: DevWorkflowDependencies,
  config: DevWorkflowConfig = defaultConfig
): Promise<DevWorkflowResult> {
  const logger = createLogger({ taskId });
  const startTime = Date.now();

  try {
    const workflow = createDevWorkflow(deps);

    const result = await workflow.invoke(
      { taskId, status: "pending" },
      {
        configurable: { thread_id: taskId },
        recursionLimit: config.recursionLimit
      }
    );

    return {
      success: result.status === "complete",
      prNumber: result.prNumber,
      status: result.status,
      durationMs: Date.now() - startTime
    };
  } catch (error) {
    // Update Linear status on failure
    await updateIssueStatus(deps.linearClient, taskId, "failed");
    await emitError(deps.linearClient, taskId, error.message);

    return {
      success: false,
      status: "failed",
      error: error.message,
      durationMs: Date.now() - startTime
    };
  } finally {
    // ALWAYS clean up sandbox
    await deps.sandbox.cleanup();
  }
}
```
