**runTestsNode:**
Takes sandbox instance as dependency injection. Writes all files from state to sandbox, runs tests, returns result.

```typescript
function createRunTestsNode(sandbox: Sandbox) {
  return async function runTestsNode(state: DevWorkflowState): Promise<Partial<DevWorkflowState>> {
    // Write all files to sandbox
    for (const file of state.files) {
      await sandbox.writeFile(file.path, file.content);
    }

    // Run tests
    const testResult = await sandbox.runTests(config.testCommand);

    return {
      testResult,
      testAttempts: state.testAttempts + 1,
      status: testResult.passed ? "committing" : "fixing"
    };
  };
}
```

**fixCodeNode:**
Uses LLM with structured output to generate fixes based on test failures.

```typescript
async function fixCodeNode(state: DevWorkflowState): Promise<Partial<DevWorkflowState>> {
  const llm = new ChatAnthropic({ model: "claude-sonnet-4-20250514" });
  const structuredLlm = llm.withStructuredOutput(CodeGenerationOutputSchema);

  const prompt = `
The tests failed. Fix the code.

Task: ${state.taskDescription}

Test output:
stdout: ${state.testResult?.stdout}
stderr: ${state.testResult?.stderr}

Current files:
${state.files.map(f => `### ${f.path}\n\`\`\`\n${f.content}\n\`\`\``).join('\n\n')}

Generate corrected files. Make minimal changes to fix the specific error.
Focus on the test failure, not on adding unrelated improvements.
`;

  const result = await structuredLlm.invoke(prompt);

  return {
    files: result.files,
    status: "testing"
  };
}
```

**Routing function:**
```typescript
function routeAfterTest(state: DevWorkflowState, config: DevWorkflowConfig): "fix_code" | "commit" | "fail" {
  if (state.testResult?.passed) return "commit";
  if (state.testAttempts >= config.maxTestAttempts) return "fail";
  return "fix_code";
}
```

**StateGraph structure:**
```typescript
const workflow = new StateGraph(DevWorkflowState)
  .addNode("generate_code", generateCodeNode)
  .addNode("run_tests", runTestsNode)
  .addNode("fix_code", fixCodeNode)
  .addEdge("__start__", "generate_code")
  .addEdge("generate_code", "run_tests")
  .addConditionalEdges("run_tests", routeAfterTest, {
    fix_code: "fix_code",
    commit: "__end__",
    fail: "__end__"
  })
  .addEdge("fix_code", "run_tests");
```
