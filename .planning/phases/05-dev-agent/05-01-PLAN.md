**generateCodeNode function:**
```typescript
async function generateCodeNode(state: DevWorkflowState): Promise<Partial<DevWorkflowState>> {
  const llm = new ChatAnthropic({ model: "claude-sonnet-4-20250514" });
  const structuredLlm = llm.withStructuredOutput(CodeGenerationOutputSchema);

  const prompt = buildCodeGenPrompt(state.taskDescription);
  const result = await structuredLlm.invoke(prompt);

  return {
    files: result.files,
    status: "testing" as const
  };
}
```
