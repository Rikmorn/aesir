/**
 * Linear API Mock Handlers
 *
 * Default responses for Linear GraphQL API.
 */

import { http, HttpResponse } from "msw";

/**
 * Default Linear GraphQL responses.
 * Tests can override these with server.use() for specific scenarios.
 */
export const linearHandlers = [
  // GraphQL endpoint - handles all Linear API requests
  http.post("https://api.linear.app/graphql", async ({ request }) => {
    const body = await request.json() as { query: string; variables?: Record<string, unknown> };
    const { query } = body;

    // Issue query
    if (query.includes("issue(id:") || query.includes("issue(")) {
      return HttpResponse.json({
        data: {
          issue: {
            id: "test_issue_id",
            identifier: "TEST-1",
            title: "Test Issue",
            description: "Test issue description",
            priority: 2,
            state: {
              id: "state_todo",
              name: "Todo",
            },
            team: {
              id: "team_test",
              name: "Test Team",
            },
            url: "https://linear.app/test/issue/TEST-1",
            createdAt: "2024-01-01T00:00:00.000Z",
            updatedAt: "2024-01-01T00:00:00.000Z",
          },
        },
      });
    }

    // Teams query
    if (query.includes("teams")) {
      return HttpResponse.json({
        data: {
          teams: {
            nodes: [
              {
                id: "team_test",
                name: "Test Team",
                key: "TEST",
              },
            ],
          },
        },
      });
    }

    // Issue create mutation
    if (query.includes("issueCreate")) {
      return HttpResponse.json({
        data: {
          issueCreate: {
            success: true,
            issue: {
              id: "new_issue_id",
              identifier: "TEST-99",
              title: "New Issue",
              url: "https://linear.app/test/issue/TEST-99",
            },
          },
        },
      });
    }

    // Issue update mutation
    if (query.includes("issueUpdate")) {
      return HttpResponse.json({
        data: {
          issueUpdate: {
            success: true,
            issue: {
              id: "test_issue_id",
              identifier: "TEST-1",
              title: "Updated Issue",
            },
          },
        },
      });
    }

    // Default: return empty data
    return HttpResponse.json({ data: {} });
  }),
];
```

/**

 *
 * Default responses for GitHub REST API.

import { http, HttpResponse } from "msw";

const GITHUB_API = "https://api.github.com";

/**
 * Default GitHub REST API responses.
 * Tests can override these with server.use() for specific scenarios.
 */
export const githubHandlers = [
  // Get repository
  http.get(`${GITHUB_API}/repos/:owner/:repo`, ({ params }) => {
    return HttpResponse.json({
      id: 12345,
      name: params.repo,
      full_name: `${params.owner}/${params.repo}`,
      private: false,
      html_url: `https://github.com/${params.owner}/${params.repo}`,
      default_branch: "main",
    });
  }),

  // List pull requests
  http.get(`${GITHUB_API}/repos/:owner/:repo/pulls`, () => {
    return HttpResponse.json([
      {
        id: 1,
        number: 1,
        title: "Test PR",
        state: "open",
        html_url: "https://github.com/test/repo/pull/1",
        head: { ref: "feature/test" },
        base: { ref: "main" },
      },
    ]);
  }),

  // Get pull request
  http.get(`${GITHUB_API}/repos/:owner/:repo/pulls/:pull_number`, ({ params }) => {
    return HttpResponse.json({
      id: Number(params.pull_number),
      number: Number(params.pull_number),
      title: `Test PR #${params.pull_number}`,
      body: "PR description",
      state: "open",
      html_url: `https://github.com/${params.owner}/${params.repo}/pull/${params.pull_number}`,
      head: { ref: "feature/test", sha: "abc123" },
      base: { ref: "main" },
      merged: false,
    });
  }),

  // Create pull request
  http.post(`${GITHUB_API}/repos/:owner/:repo/pulls`, async ({ params, request }) => {
    const body = await request.json() as { title: string; body?: string; head: string; base: string };
    return HttpResponse.json(
      {
        id: 99,
        number: 99,
        title: body.title,
        body: body.body ?? "",
        state: "open",
        html_url: `https://github.com/${params.owner}/${params.repo}/pull/99`,
        head: { ref: body.head },
        base: { ref: body.base },
      },
      { status: 201 }
    );
  }),

  // Create/update file (commits)
  http.put(`${GITHUB_API}/repos/:owner/:repo/contents/:path`, async ({ params, request }) => {
    const body = await request.json() as { message: string; content: string; sha?: string };
    return HttpResponse.json({
      content: {
        name: String(params.path).split("/").pop(),
        path: params.path,
        sha: "newsha123",
      },
      commit: {
        sha: "commit123",
        message: body.message,
      },
    });
  }),

  // Get file contents
  http.get(`${GITHUB_API}/repos/:owner/:repo/contents/:path`, ({ params }) => {
    return HttpResponse.json({
      name: String(params.path).split("/").pop(),
      path: params.path,
      sha: "sha123",
      content: Buffer.from("file content").toString("base64"),
      encoding: "base64",
    });
  }),

  // Create branch (via refs)
  http.post(`${GITHUB_API}/repos/:owner/:repo/git/refs`, async ({ request }) => {
    const body = await request.json() as { ref: string; sha: string };
    return HttpResponse.json(
      {
        ref: body.ref,
        object: { sha: body.sha },
      },
      { status: 201 }
    );
  }),

  // Get branch
  http.get(`${GITHUB_API}/repos/:owner/:repo/branches/:branch`, ({ params }) => {
    return HttpResponse.json({
      name: params.branch,
      commit: { sha: "headsha123" },
    });
  }),
];

 * Default Slack Web API responses.

  http.post(`${SLACK_API}/chat.postMessage`, async ({ request }) => {
    const formData = await request.formData();
    const channel = formData.get("channel");
    return HttpResponse.json({
      ok: true,
      channel: channel ?? "C12345",
      ts: "1234567890.123456",
      message: {
        type: "message",
        ts: "1234567890.123456",
      },
    });
  }),

  // Get conversations.history
  http.get(`${SLACK_API}/conversations.history`, () => {
    return HttpResponse.json({
      ok: true,
      messages: [
        {
          type: "message",
          ts: "1234567890.123456",
          text: "Test message",
          user: "U12345",
        },

      has_more: false,
    });
  }),

/**
 * MSW Server Setup
 *
 * Pre-configured server for mocking external APIs in Node.js tests.
 *
 * Usage in vitest.setup.ts or test files:
 * ```typescript
 * import { server } from "@aesir/test-utils";
 * import { beforeAll, afterEach, afterAll } from "vitest";
 *
 * beforeAll(() => server.listen());
 * afterEach(() => server.resetHandlers());
 * afterAll(() => server.close());
 * ```
 *
 * To override handlers in a specific test:
 * ```typescript
 * import { http, HttpResponse } from "msw";
 *
 * it("should handle error", async () => {
 *   server.use(
 *     http.get("https://api.github.com/repos/:owner/:repo", () => {
 *       return HttpResponse.json({ message: "Not Found" }, { status: 404 });
 *     })
 *   );
 *   // Test error handling...
 * });
 * ```
 */

import { setupServer } from "msw/node";
import { allHandlers } from "./handlers/index.js";

/**
 * MSW server instance with all default handlers pre-loaded.
 *
 * IMPORTANT: Call server.listen() in beforeAll and server.close() in afterAll.
 * Call server.resetHandlers() in afterEach to prevent handler pollution.
 */
export const server = setupServer(...allHandlers);

/**
 * Setup function for vitest.setup.ts files.
 * Automatically configures beforeAll/afterEach/afterAll hooks.
 *
 * @param hooks - Vitest hook functions
 */
export function setupMSW(hooks: {
  beforeAll: (fn: () => void) => void;
  afterEach: (fn: () => void) => void;
  afterAll: (fn: () => void) => void;
}): void {
  hooks.beforeAll(() => {
    server.listen({ onUnhandledRequest: "bypass" });
  });
  hooks.afterEach(() => {
    server.resetHandlers();
  });
  hooks.afterAll(() => {
    server.close();
  });
}
```
