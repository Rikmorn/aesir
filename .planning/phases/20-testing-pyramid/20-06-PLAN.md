/**
 * Linear Schema Migration SQL
 *
 * SQL to create the linear.* schema for integration tests.
 * This mirrors the production schema from @aesir/integration-linear.
 */

export const linearMigrationSql = `
-- Create linear schema
CREATE SCHEMA IF NOT EXISTS linear;

-- Credentials table
CREATE TABLE IF NOT EXISTS linear.credentials (
  id VARCHAR(32) PRIMARY KEY,
  workspace_id VARCHAR(255) NOT NULL,
  access_token TEXT NOT NULL,
  refresh_token TEXT,
  token_type VARCHAR(50) DEFAULT 'Bearer',
  scope TEXT,
  expires_at TIMESTAMP WITH TIME ZONE,
  created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  deleted_at TIMESTAMP WITH TIME ZONE
);

-- Unique constraint on workspace (only one active credential per workspace)
CREATE UNIQUE INDEX IF NOT EXISTS linear_credentials_workspace_unique
  ON linear.credentials (workspace_id)
  WHERE deleted_at IS NULL;

-- Processed deliveries for idempotency
CREATE TABLE IF NOT EXISTS linear.processed_deliveries (
  delivery_id VARCHAR(255) PRIMARY KEY,
  processed_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW()
);

-- MCP tool permissions
CREATE TABLE IF NOT EXISTS linear.mcp_tool_permissions (
  id VARCHAR(32) PRIMARY KEY,
  agent_id VARCHAR(255) NOT NULL,
  tool_name VARCHAR(255) NOT NULL,
  allowed BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  UNIQUE(agent_id, tool_name)
);

-- Updated_at trigger function
CREATE OR REPLACE FUNCTION linear.update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Apply trigger to credentials
DROP TRIGGER IF EXISTS update_linear_credentials_updated_at ON linear.credentials;
CREATE TRIGGER update_linear_credentials_updated_at
  BEFORE UPDATE ON linear.credentials
  FOR EACH ROW
  EXECUTE FUNCTION linear.update_updated_at();

-- Apply trigger to mcp_tool_permissions
DROP TRIGGER IF EXISTS update_linear_mcp_permissions_updated_at ON linear.mcp_tool_permissions;
CREATE TRIGGER update_linear_mcp_permissions_updated_at
  BEFORE UPDATE ON linear.mcp_tool_permissions
  FOR EACH ROW
  EXECUTE FUNCTION linear.update_updated_at();
`;
```

/**
 * Linear Credential Store Integration Tests
 *
 * Tests the credential store against a real PostgreSQL database.
 * Uses testcontainers for isolated database and transactions for test isolation.
 *
 * IMPORTANT: This file is named *.integration.test.ts to be excluded from test:fast.
 * Run with: pnpm test:integration
 */

import { afterAll, afterEach, beforeAll, beforeEach, describe, expect, it } from "vitest";
import {
  setupPostgresContainer,
  cleanupPostgresContainer,
  startTestTransaction,
  linearMigrationSql,
  createMockLogger,
  type PostgresContainerContext,
  type TransactionContext,
} from "@aesir/test-utils";
import { drizzle, type NodePgDatabase } from "drizzle-orm/node-postgres";
import pg from "pg";
import { createLinearCredentialStore, type LinearCredentialStore } from "./credential-store.js";

// Mock config to prevent environment validation
vi.mock("../types/config.js", () => ({
  env: {
    LINEAR_CLIENT_ID: "test-client-id",
    LINEAR_CLIENT_SECRET: "test-client-secret",
    LINEAR_WEBHOOK_SECRET: "test-webhook-secret",
    CREDENTIAL_ENCRYPTION_KEY: "0".repeat(64),
  },
  config: {
    clientId: "test-client-id",
    clientSecret: "test-client-secret",
    webhookSecret: "test-webhook-secret",
    credentialEncryptionKey: "0".repeat(64),
  },
}));

describe("LinearCredentialStore Integration", () => {
  // Container context - shared across all tests in this suite
  let containerCtx: PostgresContainerContext;
  let pool: pg.Pool;

  // Transaction context - per-test isolation
  let txCtx: TransactionContext;
  let store: LinearCredentialStore;

  // Start container once for all tests (60 second timeout for startup)
  beforeAll(async () => {
    containerCtx = await setupPostgresContainer();

    // Create pg Pool for drizzle (drizzle needs node-postgres, not postgres.js)
    pool = new pg.Pool({ connectionString: containerCtx.connectionUri });

    // Run migrations
    await containerCtx.sql.unsafe(linearMigrationSql);
  }, 60000);

  afterAll(async () => {
    await pool.end();
    await cleanupPostgresContainer(containerCtx);
  });

  // Transaction per test for isolation
  beforeEach(async () => {
    txCtx = await startTestTransaction(containerCtx.sql);

    // Create a drizzle instance for the store
    // Note: We need to use the transaction connection
    const db = drizzle(pool) as unknown as NodePgDatabase;

    store = createLinearCredentialStore({
      db,
      logger: createMockLogger(),
    });
  });

  afterEach(async () => {
    await txCtx.rollback();
  });

  describe("store()", () => {
    it("should store a new credential", async () => {
      const result = await store.store({
        workspaceId: "ws_test_1",
        accessToken: "test_access_token",
        refreshToken: "test_refresh_token",
      });

      expect(result.isOk()).toBe(true);
      if (result.isOk()) {
        expect(result.value.workspaceId).toBe("ws_test_1");
        expect(result.value.accessToken).toBeDefined(); // Encrypted
        expect(result.value.id).toMatch(/^cred_/);
      }
    });

    it("should soft-delete existing credential when storing new one", async () => {
      // Store first credential
      const first = await store.store({
        workspaceId: "ws_test_2",
        accessToken: "token_1",
      });
      expect(first.isOk()).toBe(true);

      // Store second credential for same workspace
      const second = await store.store({
        workspaceId: "ws_test_2",
        accessToken: "token_2",
      });
      expect(second.isOk()).toBe(true);

      // Get should return the new one
      const current = await store.getByWorkspace("ws_test_2");
      expect(current.isOk()).toBe(true);
      if (current.isOk() && current.value) {
        expect(current.value.accessToken).not.toBe("token_1");
      }
    });
  });

  describe("get()", () => {
    it("should return null for non-existent credential", async () => {
      const result = await store.get("cred_nonexistent");

      expect(result.isOk()).toBe(true);
      if (result.isOk()) {
        expect(result.value).toBeNull();
      }
    });

    it("should retrieve stored credential by ID", async () => {
      // First store a credential
      const stored = await store.store({
        workspaceId: "ws_test_3",
        accessToken: "test_token",
      });

      expect(stored.isOk()).toBe(true);
      if (!stored.isOk()) return;

      // Then retrieve it
      const result = await store.get(stored.value.id);

      expect(result.isOk()).toBe(true);
      if (result.isOk()) {
        expect(result.value).not.toBeNull();
        expect(result.value?.workspaceId).toBe("ws_test_3");
      }
    });
  });

  describe("getByWorkspace()", () => {
    it("should return null for workspace with no credentials", async () => {
      const result = await store.getByWorkspace("ws_empty");

      expect(result.isOk()).toBe(true);
      if (result.isOk()) {
        expect(result.value).toBeNull();
      }
    });

    it("should retrieve credential by workspace ID", async () => {
      await store.store({
        workspaceId: "ws_test_4",
        accessToken: "test_token",
      });

      const result = await store.getByWorkspace("ws_test_4");

      expect(result.isOk()).toBe(true);
      if (result.isOk()) {
        expect(result.value).not.toBeNull();
        expect(result.value?.workspaceId).toBe("ws_test_4");
      }
    });
  });

  describe("delete()", () => {
    it("should return false for non-existent credential", async () => {
      const result = await store.delete("cred_nonexistent");

      expect(result.isOk()).toBe(true);
      if (result.isOk()) {
        expect(result.value).toBe(false);
      }
    });

    it("should soft-delete existing credential", async () => {
      const stored = await store.store({
        workspaceId: "ws_test_5",
        accessToken: "test_token",
      });

      expect(stored.isOk()).toBe(true);
      if (!stored.isOk()) return;

      const deleted = await store.delete(stored.value.id);
      expect(deleted.isOk()).toBe(true);
      if (deleted.isOk()) {
        expect(deleted.value).toBe(true);
      }

      // Should no longer be retrievable
      const result = await store.get(stored.value.id);
      expect(result.isOk()).toBe(true);
      if (result.isOk()) {
        expect(result.value).toBeNull();
      }
    });
  });

  describe("test isolation", () => {
    it("should start with clean state (no leftover data from other tests)", async () => {
      // This test runs after previous tests that created credentials
      // Transaction rollback should have cleaned them up
      const result = await store.getByWorkspace("ws_test_1");

      expect(result.isOk()).toBe(true);
      if (result.isOk()) {
        // Should be null because previous test's data was rolled back
        expect(result.value).toBeNull();
      }
    });
  });
});
```

Note: This test demonstrates:
1. Container setup in beforeAll (one-time)
2. Transaction isolation in beforeEach/afterEach (per-test)
3. Real database operations with proper error handling
4. Verification that test isolation works (last test case)

The test may need adjustment based on how credential-store.ts handles the drizzle connection.
