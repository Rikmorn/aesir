**Gap 2 Fix: Temporal activity DI**

The problem: Activities are defined with client as first parameter, but workflows can't serialize/pass clients. The workflow passes `{}` which fails at runtime.

The solution: Create a factory function that binds clients at worker startup time.

1. In `src/temporal/activities/index.ts`:
   - Add imports for client types:
   ```typescript
   import type { WebClient } from "@slack/web-api";
   import type { Octokit } from "@octokit/rest";
   import type { LinearClient } from "@linear/sdk";
   import type { Sandbox } from "../../sandbox/types.js";
   ```

   - Create `ActivityDependencies` interface:
   ```typescript
   export interface ActivityDependencies {
     slackClient: WebClient;
     octokit: Octokit;
     linearClient: LinearClient;
     sandbox: Sandbox;
   }
   ```

   - Create `makeActivities` factory function:
   ```typescript
   export function makeActivities(deps: ActivityDependencies) {
     return {
       // Bind clients at creation time - workflow just passes remaining args
       sendApprovalRequestActivity: (
         notification: Parameters<typeof sendApprovalRequestActivity>[1],
         channel: Parameters<typeof sendApprovalRequestActivity>[2]
       ) => sendApprovalRequestActivity(deps.slackClient, notification, channel),

       sendStatusUpdateActivity: (
         notification: Parameters<typeof sendStatusUpdateActivity>[1],
         channel: Parameters<typeof sendStatusUpdateActivity>[2]
       ) => sendStatusUpdateActivity(deps.slackClient, notification, channel),

       mergePRActivity: (
         input: Parameters<typeof mergePRActivity>[1]
       ) => mergePRActivity(deps.octokit, input),

       updateLinearStatusActivity: (
         issueId: Parameters<typeof updateLinearStatusActivity>[1],
         statusName: Parameters<typeof updateLinearStatusActivity>[2]
       ) => updateLinearStatusActivity(deps.linearClient, issueId, statusName),

       executeDevWorkflow: (
         taskId: string
       ) => executeDevWorkflow(taskId, {
         linearClient: deps.linearClient,
         octokit: deps.octokit,
         sandbox: deps.sandbox,
         // Config can use defaults or be passed in
       }),
     };
   }
   ```

   - Export the types for the bound activities:
   ```typescript
   export type BoundActivities = ReturnType<typeof makeActivities>;
   ```

2. In `src/temporal/worker.ts`:
   - Update imports:
   ```typescript
   import { makeActivities, type ActivityDependencies } from "./activities/index.js";
   ```

   - Add `dependencies` to `WorkerConfig`:
   ```typescript
   export interface WorkerConfig {
     address?: string;
     namespace?: string;
     taskQueue: string;
     dependencies?: ActivityDependencies;  // ADD THIS
   }
   ```

   - Update `createTemporalWorker` to use factory:
   ```typescript
   // Replace the dynamic require with factory pattern
   let activities: object = {};
   if (config.dependencies) {
     activities = makeActivities(config.dependencies);
     logger.info("temporal_worker_activities_bound", {
       message: "Activities bound with dependencies",
     });
   } else {
     logger.warn("temporal_worker_no_dependencies", {
       message: "No dependencies provided, activities will fail at runtime",
     });
   }
   ```

3. Note: The workflow file (`approval-workflow.ts`) does NOT need changes. The proxied activities will now receive bound functions that don't need the first client parameter. The `{} as Parameters<...>[0]` calls become harmless since the bound functions ignore that parameter.
  </action>
