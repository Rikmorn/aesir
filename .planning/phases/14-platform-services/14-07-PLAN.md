```typescript
  executionTracker: ExecutionTracker;
  workspaceId: string;
}

export async function linearWebhookHandler(
  req: WebhookRequest,
  res: WebhookResponse,
  config: LinearWebhookConfig,
  secret: string,
  services?: WebhookServices  // Optional for backward compatibility during transition
): Promise<void> {
```

2. Add idempotency check FIRST after signature verification:
```typescript
// After signature verification, BEFORE any processing
if (services?.webhookIdempotency) {
  const deliveryId = req.headers["linear-delivery"];
  if (deliveryId) {
    const { isDuplicate } = await services.webhookIdempotency.checkAndRecord(
      "linear",
      deliveryId,
      payload.type ?? "unknown"
    );

    if (isDuplicate) {
      res.status(200).json({
        success: true,
        message: "Duplicate webhook - already processed",
      });
      // Set header for duplicate indication
      // Note: The res adapter may need updating to support headers
      return;
    }
  }
}
```

3. Wrap agent processing with execution tracking:
```typescript
let executionId: string | undefined;

// Start execution tracking
if (services?.executionTracker && issueId) {
  executionId = await services.executionTracker.start({
    agentType: "dev-agent",
    issueId,
    workspaceId: services.workspaceId,
  });
}

try {
  // Existing agent processing logic...
  const result = await handleAgentSessionWebhook(payload, config);

  // Mark execution complete
  if (executionId && services?.executionTracker) {
    await services.executionTracker.complete(executionId);
  }

  res.status(200).json({ success: true, result });
} catch (error) {
  // Mark execution failed
  if (executionId && services?.executionTracker) {
    const lastState = error instanceof Error ? error.message : "unknown";
    await services.executionTracker.fail(executionId, lastState);
  }
  throw error;
}
```

Key points:
- Idempotency check is FIRST after signature verification
- Services are optional for backward compatibility during transition
- Execution tracking wraps the actual work
- Failed executions capture error message as last_known_state
