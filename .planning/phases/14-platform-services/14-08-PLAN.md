### Dependency Injection Pattern

Services use factory functions with options objects for dependency injection:

```typescript
import type { PinoLogger } from "@aesir/common";
import type { PostgresJsDatabase } from "drizzle-orm/postgres-js";

interface MyServiceOptions {
  db: PostgresJsDatabase;
  logger: PinoLogger;
  config?: { someOption: boolean };
}

interface MyService {
  doSomething(): Promise<void>;
  health(): Promise<{ healthy: boolean; latencyMs: number }>;
  close(): Promise<void>;
}

export function createMyService(options: MyServiceOptions): MyService {
  const { db, logger, config } = options;

  if (!db) throw new Error("db is required");
  if (!logger) throw new Error("logger is required");

  return {
    async doSomething() {
      // Implementation using db and logger
    },
    async health() {
      const start = Date.now();
      await db.execute(sql`SELECT 1`);
      return { healthy: true, latencyMs: Date.now() - start };
    },
    async close() {
      // Cleanup if needed
    },
  };
}
```

**Rules:**
- Services are created at startup in main.ts (not per-request)
- Dependencies passed via options object (db, logger, config)
- Throw on invalid config (fail fast)
- Include health() and close() methods on all services
- No global singletons - services are explicit dependencies
- Use child loggers with service context: `logger.child({ service: "my-service" })`

**Testing:**
```typescript
describe("MyService", () => {
  it("should do something", async () => {
    const mockDb = { execute: vi.fn() } as unknown as PostgresJsDatabase;
    const mockLogger = { info: vi.fn(), error: vi.fn() } as unknown as PinoLogger;

    const service = createMyService({ db: mockDb, logger: mockLogger });
    await service.doSomething();

    expect(mockDb.execute).toHaveBeenCalled();
  });
});
```
```

This documents the pattern so future code follows the same approach.
