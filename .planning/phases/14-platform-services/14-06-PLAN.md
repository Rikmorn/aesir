```typescript
/**
 * Sync Cursor Service
 *
 * Manages synchronization cursors for incremental data syncs.
 * Tracks last sync position per workspace+provider+resource combination.
 */

import type { PinoLogger } from "@aesir/common";
import { eq, and, sql } from "drizzle-orm";
import type { PostgresJsDatabase } from "drizzle-orm/postgres-js";
import { syncCursors } from "../db/schema.js";

export type SyncProvider = "linear" | "github" | "slack";
export type SyncResourceType = "issues" | "comments" | "pull_requests" | "messages";

export interface SyncCursorServiceOptions {
  db: PostgresJsDatabase;
  logger: PinoLogger;
}

export interface SyncCursorKey {
  workspaceId: string;
  provider: SyncProvider;
  resourceType: SyncResourceType;
}

export interface SyncCursorValue {
  cursor: string | null;
  lastSyncAt: Date | null;
}

export interface SyncCursorService {
  /**
   * Get the current sync cursor for a workspace+provider+resource
   */
  get(key: SyncCursorKey): Promise<SyncCursorValue | null>;

  /**
   * Set/update the sync cursor
   * Creates if not exists, updates if exists (upsert)
   */
  set(key: SyncCursorKey, value: { cursor: string }): Promise<void>;

  /**
   * Clear (delete) a sync cursor
   * Used when resetting sync state
   */
  clear(key: SyncCursorKey): Promise<void>;

  /**
   * Health check
   */
  health(): Promise<{ healthy: boolean; latencyMs: number }>;

  /**
   * Cleanup resources
   */
  close(): Promise<void>;
}

/**
 * Create sync cursor service
 */
export function createSyncCursorService(
  options: SyncCursorServiceOptions
): SyncCursorService {
  const { db, logger } = options;

  if (!db) throw new Error("db is required for SyncCursorService");
  if (!logger) throw new Error("logger is required for SyncCursorService");

  return {
    async get(key: SyncCursorKey): Promise<SyncCursorValue | null> {
      const { workspaceId, provider, resourceType } = key;

      const [row] = await db
        .select({
          cursor: syncCursors.cursor,
          lastSyncAt: syncCursors.last_sync_at,
        })
        .from(syncCursors)
        .where(
          and(
            eq(syncCursors.workspace_id, workspaceId),
            eq(syncCursors.provider, provider),
            eq(syncCursors.resource_type, resourceType)
          )
        );

      if (!row) {
        return null;
      }

      return {
        cursor: row.cursor,
        lastSyncAt: row.lastSyncAt,
      };
    },

    async set(key: SyncCursorKey, value: { cursor: string }): Promise<void> {
      const { workspaceId, provider, resourceType } = key;
      const now = new Date();

      // Upsert: insert or update on conflict
      await db
        .insert(syncCursors)
        .values({
          workspace_id: workspaceId,
          provider,
          resource_type: resourceType,
          cursor: value.cursor,
          last_sync_at: now,
          updated_at: now,
        })
        .onConflictDoUpdate({
          target: [
            syncCursors.workspace_id,
            syncCursors.provider,
            syncCursors.resource_type,
          ],
          set: {
            cursor: value.cursor,
            last_sync_at: now,
            updated_at: now,
          },
        });

      logger.debug(
        { workspaceId, provider, resourceType, cursor: value.cursor },
        "Sync cursor updated"
      );
    },

    async clear(key: SyncCursorKey): Promise<void> {
      const { workspaceId, provider, resourceType } = key;

      await db
        .delete(syncCursors)
        .where(
          and(
            eq(syncCursors.workspace_id, workspaceId),
            eq(syncCursors.provider, provider),
            eq(syncCursors.resource_type, resourceType)
          )
        );

      logger.info(
        { workspaceId, provider, resourceType },
        "Sync cursor cleared"
      );
    },

    async health(): Promise<{ healthy: boolean; latencyMs: number }> {
      const start = Date.now();
      try {
        await db.execute(sql`SELECT 1`);
        return { healthy: true, latencyMs: Date.now() - start };
      } catch {
        return { healthy: false, latencyMs: Date.now() - start };
      }
    },

    async close(): Promise<void> {
      // No resources to clean up
    },
  };
}
```

Key implementation notes:
- Uses upsert (onConflictDoUpdate) for set operation
- Tracks last_sync_at timestamp on each update
- clear() allows resetting sync state for re-sync scenarios
- Resource types are string-typed for flexibility
