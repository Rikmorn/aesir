```typescript
export async function orchestratorWorkflow(
  input: OrchestratorWorkflowInput,
): Promise<OrchestratorWorkflowResult> {
  const { taskId, issue, slackChannel } = input;
  const workflowId = wf.workflowInfo().workflowId;

  // Mutable state for signal handlers
  const state = {
    phase: "pending" as OrchestratorWorkflowPhase,
    approval: null as PlanApprovalPayload | null,
    prFeedback: null as string | null,
    prCompletion: null as PRCompletionPayload | null,
    escalationResolution: null as { action: "retry" | "abort"; guidance?: string } | null,
    prNumber: undefined as number | undefined,
    prUrl: undefined as string | undefined,
    errorMessage: undefined as string | undefined,
    totalTokenCount: { input: 0, output: 0 },
  };

  // Accumulate token counts across activities
  function addTokens(count: { input: number; output: number }): void {
    state.totalTokenCount.input += count.input;
    state.totalTokenCount.output += count.output;
  }

  // --- Signal handlers ---
  wf.setHandler(planApprovalSignal, (decision) => {
    wf.log.info("Received plan approval signal", { approved: decision.approved });
    state.approval = decision;
  });

  wf.setHandler(prFeedbackSignal, (feedback) => {
    wf.log.info("Received PR feedback signal", { feedbackLength: feedback.length });
    state.prFeedback = feedback;
  });

  wf.setHandler(prCompletionSignal, (completion) => {
    wf.log.info("Received PR completion signal", { merged: completion.merged, prNumber: completion.prNumber });
    state.prCompletion = completion;
  });

  wf.setHandler(escalationResolvedSignal, (resolution) => {
    wf.log.info("Received escalation resolution signal", { action: resolution.action });
    state.escalationResolution = resolution;
  });

  // --- Query handler ---
  wf.setHandler(orchestratorStatusQuery, (): OrchestratorQueryStatus => ({
    taskId,
    phase: state.phase,
    prNumber: state.prNumber,
    prUrl: state.prUrl,
    errorMessage: state.errorMessage,
  }));

  // === Phase 1: Setup ===
  state.phase = "setup";
  wf.log.info("Setting up container", { taskId, issueIdentifier: issue.identifier });

  try {
    await infrastructureActivities.setupContainerActivity({
      taskId,
      issue: { identifier: issue.identifier, title: issue.title },
      workflowId,
    });
  } catch (error) {
    state.phase = "failed";
    state.errorMessage = `Container setup failed: ${String(error)}`;
    await wf.condition(wf.allHandlersFinished);
    return { success: false, phase: "failed", errorMessage: state.errorMessage };
  }

  // === Phase 2: Pre-approval loop ===
  let approved = false;
  let rejectionFeedback: string | undefined;

  while (!approved) {
    state.phase = "pre_approval";
    wf.log.info("Running pre-approval orchestrator", { taskId, hasRejectionFeedback: !!rejectionFeedback });

    const preApprovalInput: { taskId: string; issue: typeof issue; slackChannel: string; workflowId: string; rejectionFeedback?: string } = {
      taskId,
      issue,
      slackChannel,
      workflowId,
    };
    if (rejectionFeedback !== undefined) {
      preApprovalInput.rejectionFeedback = rejectionFeedback;
    }

    let preResult;
    try {
      preResult = await orchestratorActivities.runOrchestratorPreApproval(preApprovalInput);
    } catch (error) {
      state.phase = "failed";
      state.errorMessage = `Pre-approval failed: ${String(error)}`;
      await infrastructureActivities.stopContainerActivity(taskId).catch(() => {});
      await wf.condition(wf.allHandlersFinished);
      return { success: false, phase: "failed", errorMessage: state.errorMessage };
    }

    addTokens(preResult.tokenCount);

    if (preResult.status === "error") {
      state.phase = "failed";
      state.errorMessage = preResult.plan;
      await infrastructureActivities.stopContainerActivity(taskId).catch(() => {});
      await wf.condition(wf.allHandlersFinished);
      return { success: false, phase: "failed", errorMessage: state.errorMessage };
    }

    // === Wait for approval signal ===
    state.phase = "awaiting_approval";
    wf.log.info("Waiting for plan approval", { taskId });

    const receivedApproval = await wf.condition(
      () => state.approval !== null,
      REMINDER_TIMEOUT,
    );

    if (!receivedApproval) {
      // 24h timeout -- stop container, send reminder (orchestrator already sent approval request via tools)
      wf.log.info("Approval timeout (24h), stopping container", { taskId });
      await infrastructureActivities.stopContainerActivity(taskId).catch(() => {});

      // Wait longer (48h more = 72h total)
      const lateApproval = await wf.condition(
        () => state.approval !== null,
        FINAL_TIMEOUT,
      );

      if (!lateApproval) {
        state.phase = "timeout";
        await wf.condition(wf.allHandlersFinished);
        return { success: false, phase: "timeout" };
      }
    }

    // Process approval
    if (state.approval?.approved) {
      approved = true;
    } else {
      // Rejected -- loop back with feedback
      rejectionFeedback = state.approval?.feedback ?? "Plan rejected without specific feedback.";
      state.approval = null; // Reset for next cycle
      wf.log.info("Plan rejected, re-planning", { feedback: rejectionFeedback });
    }
  }

  // === Phase 3: Post-approval execution ===
  state.phase = "post_approval";
  wf.log.info("Running post-approval orchestrator", { taskId });

  let postResult;
  try {
    postResult = await orchestratorActivities.runOrchestratorPostApproval({
      taskId,
      issue,
      slackChannel,
      workflowId,
    });
  } catch (error) {
    state.phase = "failed";
    state.errorMessage = `Post-approval failed: ${String(error)}`;
    await infrastructureActivities.stopContainerActivity(taskId).catch(() => {});
    await wf.condition(wf.allHandlersFinished);
    return { success: false, phase: "failed", errorMessage: state.errorMessage };
  }

  addTokens(postResult.tokenCount);

  if (postResult.status === "error") {
    state.phase = "failed";
    state.errorMessage = postResult.errorMessage ?? "Post-approval orchestrator failed";
    await infrastructureActivities.stopContainerActivity(taskId).catch(() => {});
    await wf.condition(wf.allHandlersFinished);
    return { success: false, phase: "failed", errorMessage: state.errorMessage };
  }

  // Update state with PR info
  state.prNumber = postResult.prNumber;
  state.prUrl = postResult.prUrl;

  // === Phase 4: PR wait + feedback loop ===
  if (state.prUrl) {
    state.phase = "awaiting_pr";
    wf.log.info("PR created, waiting for feedback or completion", {
      prNumber: state.prNumber,
      prUrl: state.prUrl,
    });

    while (true) {
      const receivedSignal = await wf.condition(
        () => state.prCompletion !== null || state.prFeedback !== null,
        FEEDBACK_TIMEOUT,
      );

      // PR completion (merge or close)
      if (state.prCompletion) {
        if (state.prCompletion.merged) {
          state.phase = "complete";
          wf.log.info("PR merged", { prNumber: state.prCompletion.prNumber });
        } else {
          state.phase = "failed";
          state.errorMessage = "PR closed without merging";
          wf.log.info("PR closed without merge", { prNumber: state.prCompletion.prNumber });
        }
        break;
      }

      // PR feedback -- address and continue waiting
      if (receivedSignal && state.prFeedback) {
        state.phase = "addressing_feedback";
        wf.log.info("Addressing PR feedback", { taskId });

        try {
          const feedbackResult = await orchestratorActivities.handleOrchestratorFeedback({
            taskId,
            issue,
            slackChannel,
            workflowId,
            feedback: state.prFeedback,
          });
          addTokens(feedbackResult.tokenCount);
        } catch (error) {
          wf.log.warn("Feedback handling failed", { error: String(error) });
          // Continue waiting -- don't fail the workflow on feedback error
        }

        state.prFeedback = null; // Reset for next round
        state.phase = "awaiting_pr";
        continue;
      }

      // Timeout -- no signals received within 7 days
      wf.log.info("PR feedback timeout, completing workflow");
      state.phase = "complete";
      break;
    }
  } else {
    // No PR created but orchestrator completed -- unusual but handle gracefully
    state.phase = "complete";
    wf.log.warn("Orchestrator completed without PR creation");
  }

  // === Phase 5: Complete ===
  const success = state.phase === "complete" && state.prUrl !== undefined;
  await infrastructureActivities.completeTaskActivity({ taskId, success });
  await infrastructureActivities.stopContainerActivity(taskId).catch(() => {});

  await wf.condition(wf.allHandlersFinished);

  const result: OrchestratorWorkflowResult = {
    success,
    phase: state.phase,
  };
  // Conditional property assignment for exactOptionalPropertyTypes
  if (state.prNumber !== undefined) { result.prNumber = state.prNumber; }
  if (state.prUrl !== undefined) { result.prUrl = state.prUrl; }
  if (state.errorMessage !== undefined) { result.errorMessage = state.errorMessage; }
  if (state.totalTokenCount.input > 0 || state.totalTokenCount.output > 0) {
    result.totalTokenCount = state.totalTokenCount;
  }
  return result;
}
```
