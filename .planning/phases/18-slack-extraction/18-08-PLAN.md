    -- Slack Integration Schema
    -- Creates the slack.* namespace for isolated Slack data storage

    -- Create schema
    CREATE SCHEMA IF NOT EXISTS slack;

    -- Installations table (OAuth token storage, matches Bolt's Installation model)
    CREATE TABLE IF NOT EXISTS slack.installations (
      id TEXT PRIMARY KEY,
      team_id TEXT NOT NULL,
      enterprise_id TEXT,
      user_id TEXT,
      is_enterprise_install BOOLEAN DEFAULT FALSE,
      encrypted_bot_token TEXT NOT NULL,
      encrypted_user_token TEXT,
      bot_id TEXT,
      bot_user_id TEXT,
      bot_scopes TEXT,
      app_id TEXT,
      token_type TEXT DEFAULT 'bot',
      created_at TIMESTAMPTZ DEFAULT NOW(),
      updated_at TIMESTAMPTZ DEFAULT NOW(),
      deleted_at TIMESTAMPTZ
    );

    -- Unique constraint on active installations (team + enterprise)
    CREATE UNIQUE INDEX IF NOT EXISTS idx_slack_installations_active
      ON slack.installations (team_id, COALESCE(enterprise_id, ''))
      WHERE deleted_at IS NULL;

    -- Index for team_id lookups
    CREATE INDEX IF NOT EXISTS idx_slack_installations_team_id
      ON slack.installations (team_id)
      WHERE deleted_at IS NULL;

    -- Event deliveries table (for deduplication via event_id)
    CREATE TABLE IF NOT EXISTS slack.event_deliveries (
      id TEXT PRIMARY KEY,
      event_id TEXT UNIQUE NOT NULL,
      event_type TEXT NOT NULL,
      team_id TEXT,
      processed_at TIMESTAMPTZ,
      created_at TIMESTAMPTZ DEFAULT NOW()
    );

    -- Index for event_id lookups (primary deduplication key)
    CREATE INDEX IF NOT EXISTS idx_slack_event_deliveries_event_id
      ON slack.event_deliveries (event_id);

    -- Index for cleanup queries (by created_at)
    CREATE INDEX IF NOT EXISTS idx_slack_event_deliveries_created_at
      ON slack.event_deliveries (created_at);

    -- Updated_at trigger function (reuse if exists in other schemas)
    CREATE OR REPLACE FUNCTION slack.update_updated_at()
    RETURNS TRIGGER AS $$
    BEGIN
      NEW.updated_at = NOW();
      RETURN NEW;
    END;
    $$ LANGUAGE plpgsql;

    -- Apply trigger to installations table
    DROP TRIGGER IF EXISTS update_installations_updated_at ON slack.installations;
    CREATE TRIGGER update_installations_updated_at
      BEFORE UPDATE ON slack.installations
      FOR EACH ROW
      EXECUTE FUNCTION slack.update_updated_at();
    ```

    Note: Uses IF NOT EXISTS and DROP TRIGGER IF EXISTS for idempotency.

       /**
        * Slack Credential Migration Script
        *
        * Migrates existing SLACK_BOT_TOKEN from environment variable to database.
        * Run with: pnpm --filter @aesir/integration-slack migrate
        */
       import dotenvFlow from "dotenv-flow";
       import { drizzle } from "drizzle-orm/node-postgres";
       import { Pool } from "pg";
       import { nanoid } from "nanoid";
       import { createCipheriv, createDecipheriv, randomBytes } from "node:crypto";
       import * as schema from "../src/db/schema.js";

       // Load environment
       dotenvFlow.config();

       // Inline schema definition to avoid config validation
       const slackSchema = schema.slackSchema;
       const installations = schema.installations;

       // Encryption functions (inline to avoid importing config-dependent code)
       function encryptToken(token: string, encryptionKey: string | undefined): string {
         if (!encryptionKey) {
           console.warn("[migrate] No encryption key, storing token in plaintext");
           return token;
         }
         const key = Buffer.from(encryptionKey, "hex");
         const iv = randomBytes(16);
         const cipher = createCipheriv("aes-256-cbc", key, iv);
         const encrypted = Buffer.concat([cipher.update(token, "utf8"), cipher.final()]);
         return `${iv.toString("hex")}:${encrypted.toString("hex")}`;
       }

       async function migrateCredentials() {
         console.log("[migrate] Starting Slack credential migration...");

         // Get environment variables
         const botToken = process.env.SLACK_BOT_TOKEN;
         const encryptionKey = process.env.CREDENTIAL_ENCRYPTION_KEY;
         const dbHost = process.env.DB_HOST || "localhost";
         const dbPort = parseInt(process.env.DB_PORT || "5432", 10);
         const dbUser = process.env.DB_USER || "temporal";
         const dbPassword = process.env.DB_PASSWORD || "temporal";
         const dbName = process.env.DB_NAME || "temporal";

         if (!botToken) {
           console.log("[migrate] No SLACK_BOT_TOKEN found, nothing to migrate");
           return;
         }

         // Create database connection
         const pool = new Pool({
           host: dbHost,
           port: dbPort,
           user: dbUser,
           password: dbPassword,
           database: dbName,
         });

         const db = drizzle(pool, { schema });

         try {
           // Check if schema exists
           const schemaCheck = await pool.query(
             "SELECT schema_name FROM information_schema.schemata WHERE schema_name = 'slack'"
           );

           if (schemaCheck.rows.length === 0) {
             console.log("[migrate] Slack schema does not exist, running migration SQL first...");
             const fs = await import("node:fs");
             const path = await import("node:path");
             const migrationPath = path.join(
               import.meta.dirname,
               "../src/db/migrations/0000_create_slack_schema.sql"
             );
             const migrationSql = fs.readFileSync(migrationPath, "utf8");
             await pool.query(migrationSql);
             console.log("[migrate] Schema created");
           }

           // Check for existing installation
           const existing = await db
             .select()
             .from(installations)
             .where(
               sql`${installations.teamId} = 'default' AND ${installations.deletedAt} IS NULL`
             )
             .limit(1);

           if (existing.length > 0) {
             console.log("[migrate] Installation for 'default' team already exists, skipping");
             return;
           }

           // Encrypt token
           const encryptedToken = encryptToken(botToken, encryptionKey);

           // Insert new installation
           const id = `cred_${nanoid(24)}`;
           await db.insert(installations).values({
             id,
             teamId: "default",
             enterpriseId: null,
             userId: null,
             isEnterpriseInstall: false,
             encryptedBotToken: encryptedToken,
             encryptedUserToken: null,
             botId: null,
             botUserId: null,
             botScopes: null,
             appId: null,
             tokenType: "bot",
           });

           console.log(`[migrate] Successfully migrated token for team 'default' (id: ${id})`);
         } finally {
           await pool.end();
         }
       }

       migrateCredentials().catch((err) => {
         console.error("[migrate] Migration failed:", err);
         process.exit(1);
       });
       ```
