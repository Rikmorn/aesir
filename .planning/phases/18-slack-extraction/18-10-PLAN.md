         const mockLogger = {
           debug: vi.fn(),
           info: vi.fn(),
           warn: vi.fn(),
           error: vi.fn(),
           child: vi.fn(() => mockLogger),
         };

         const mockDeliveryStore = {
           isDeliveryProcessed: vi.fn(),
           recordDelivery: vi.fn(),
         };

         beforeEach(() => {
           vi.clearAllMocks();
         });

         it("should return null for duplicate events", async () => {
           mockDeliveryStore.isDeliveryProcessed.mockReturnValue(ok(true));

           const handler = createEventHandler({
             deliveryStore: mockDeliveryStore as any,
             logger: mockLogger as any,
           });

           const event = {
             type: "app_mention",
             event_id: "Ev1234567890",
             event_time: 1234567890,
             team_id: "T1234567890",
             user: "U1234567890",
             text: "hello",
             ts: "1234567890.123456",
             channel: "C1234567890",
           };

           const result = await handler.handleEvent(event);
           expect(result.isOk()).toBe(true);
           expect(result._unsafeUnwrap()).toBeNull();
         });

         it("should process new events and record delivery", async () => {
           mockDeliveryStore.isDeliveryProcessed.mockReturnValue(ok(false));
           mockDeliveryStore.recordDelivery.mockReturnValue(ok("delivery_id"));

           const handler = createEventHandler({
             deliveryStore: mockDeliveryStore as any,
             logger: mockLogger as any,
           });

           const event = {
             type: "app_mention",
             event_id: "Ev1234567890",
             event_time: 1234567890,
             team_id: "T1234567890",
             user: "U1234567890",
             text: "hello",
             ts: "1234567890.123456",
             channel: "C1234567890",
           };

           const result = await handler.handleEvent(event);
           expect(result.isOk()).toBe(true);
           const payload = result._unsafeUnwrap();
           expect(payload).not.toBeNull();
           expect(payload?.eventId).toBe("Ev1234567890");
           expect(mockDeliveryStore.recordDelivery).toHaveBeenCalled();
         });

         it("should filter bot_message subtypes", async () => {
           mockDeliveryStore.isDeliveryProcessed.mockReturnValue(ok(false));

           const handler = createEventHandler({

             logger: mockLogger as any,

           const event = {
             type: "message",
             subtype: "bot_message",

             event_time: 1234567890,
             team_id: "T1234567890",
             text: "bot says hello",
             ts: "1234567890.123456",
             channel: "C1234567890",
           };

           const result = await handler.handleEvent(event);
           expect(result.isOk()).toBe(true);
           expect(result._unsafeUnwrap()).toBeNull(); // Filtered out
         });
       });

       import { describe, expect, it } from "vitest";
       import {
         buildApprovalBlocks,
         buildStatusBlocks,
         buildSimpleMessage,
       } from "./blocks.js";

       describe("buildApprovalBlocks", () => {
         it("should create blocks with title and summary", () => {
           const blocks = buildApprovalBlocks({
             taskId: "ABC-123",
             prUrl: "https://github.com/org/repo/pull/42",
             title: "feat: Add auth",
             summary: "Implements JWT-based authentication",
           });

           expect(blocks.length).toBeGreaterThan(0);

           // Check for title section
           const titleSection = blocks.find(
             (b) => b.type === "section" && (b as any).text?.text?.includes("PR Ready for Review")
           );
           expect(titleSection).toBeDefined();
         });

         it("should include approve and reject buttons", () => {
           const blocks = buildApprovalBlocks({
             taskId: "ABC-123",
             prUrl: "https://github.com/org/repo/pull/42",
             title: "feat: Add auth",
             summary: "Implements JWT-based authentication",
           });

           // Find actions block
           const actionsBlock = blocks.find((b) => b.type === "actions");
           expect(actionsBlock).toBeDefined();

           if (actionsBlock && "elements" in actionsBlock) {
             const elements = actionsBlock.elements as any[];
             expect(elements.length).toBe(2);

             const approveButton = elements.find((e) => e.action_id === "approve_pr");
             const rejectButton = elements.find((e) => e.action_id === "reject_pr");

             expect(approveButton).toBeDefined();
             expect(approveButton.style).toBe("primary");
             expect(rejectButton).toBeDefined();
             expect(rejectButton.style).toBe("danger");
           }
         });

         it("should include task ID in button values", () => {
           const blocks = buildApprovalBlocks({
             taskId: "ABC-123",
             prUrl: "https://github.com/org/repo/pull/42",
             title: "test",
             summary: "test",
           });

           const actionsBlock = blocks.find((b) => b.type === "actions");
           if (actionsBlock && "elements" in actionsBlock) {
             const elements = actionsBlock.elements as any[];
             const approveButton = elements.find((e) => e.action_id === "approve_pr");

         });
       });
