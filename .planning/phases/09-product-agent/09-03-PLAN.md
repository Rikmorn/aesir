**graph.ts:**
```typescript
import { StateGraph, END } from "@langchain/langgraph";
import { SqliteSaver } from "@langchain/langgraph-checkpoint-sqlite";

// Graph structure:
// START -> analyze -> (route) -> clarify -> analyze (loop)
//                            -> createTasks -> END

function routeAfterAnalysis(state: ProductAgentState): string {
  if (state.phase === 'complete' || state.phase === 'creating') {
    return 'createTasks';
  }
  if (state.phase === 'clarifying') {
    return 'clarify';
  }
  // Default: need more info
  return 'clarify';
}

export function createProductAgentGraph(options: {
  llm: BaseChatModel;
  linearClient: LinearClient;
  checkpointer?: SqliteSaver;
}) {
  const graph = new StateGraph(ProductAgentStateAnnotation)
    .addNode('analyze', analyzeRequirementsNode(options))
    .addNode('clarify', generateClarificationNode(options))
    .addNode('createTasks', createTasksNode(options))
    .addEdge('__start__', 'analyze')
    .addConditionalEdges('analyze', routeAfterAnalysis)
    .addEdge('clarify', 'analyze')
    .addEdge('createTasks', '__end__');

  return graph.compile({ checkpointer: options.checkpointer });
}
```
