```typescript
// ─── Tool Registry Client ───────────────────────────────────────────────────

export async function fetchToolRegistry(): Promise<ToolRegistryEntry[]> {
  const url = `${getBaseUrl()}/api/tools/registry`;
  try {
    const response = await fetch(url, {
      signal: AbortSignal.timeout(5000),
      headers: { Accept: "application/json" },
      next: { revalidate: 60 },
    });
    if (!response.ok) {
      // biome-ignore lint/suspicious/noConsole: Server-side HTTP client
      console.error(`[agent-service] Failed to fetch tool registry: ${response.status} ${response.statusText}`);
      return [];
    }
    return (await response.json()) as ToolRegistryEntry[];
  } catch (error) {
    // biome-ignore lint/suspicious/noConsole: Server-side HTTP client
    console.error("[agent-service] Failed to fetch tool registry:", error);
    return [];
  }
}

export async function fetchToolsHealth(): Promise<IntegrationHealth[]> {
  const url = `${getBaseUrl()}/api/tools/health`;
  try {
    const response = await fetch(url, {
      signal: AbortSignal.timeout(5000),
      headers: { Accept: "application/json" },
      next: { revalidate: 30 },
    });
    if (!response.ok) {
      // biome-ignore lint/suspicious/noConsole: Server-side HTTP client
      console.error(`[agent-service] Failed to fetch tools health: ${response.status} ${response.statusText}`);
      return [];
    }
    const data = (await response.json()) as { integrations: IntegrationHealth[] };
    return data.integrations;
  } catch (error) {
    // biome-ignore lint/suspicious/noConsole: Server-side HTTP client
    console.error("[agent-service] Failed to fetch tools health:", error);
    return [];
  }
}
```

Note: `fetchToolsHealth` returns `data.integrations` because the API response shape is `{ integrations: IntegrationHealth[] }` (see `ToolsHealthResponse` in agent-service types.ts). Health data uses `revalidate: 30` (shorter than 60s) because health can change quickly.

```typescript
/**
 * Format a duration in milliseconds as a human-readable string.
 * @returns "<1ms", "Xms", "X.Xs", "Xm Xs"
 */
export function formatDurationMs(ms: number | null): string {
  if (ms === null || ms === undefined) return "-";
  if (ms < 1) return "<1ms";
  if (ms < 1000) return `${Math.round(ms)}ms`;
  const seconds = ms / 1000;
  if (seconds < 60) return `${seconds.toFixed(1)}s`;
  const minutes = Math.floor(seconds / 60);
  const remainingSeconds = Math.round(seconds % 60);
  return `${minutes}m ${remainingSeconds}s`;
}

/**
 * Format a number as a percentage string.
 * @returns "0%", "X.X%", or "100%"
 */
export function formatPercentage(value: number): string {
  if (value === 0) return "0%";
  if (value >= 1) return "100%";
  return `${(value * 100).toFixed(1)}%`;
}
```

**Functions to implement:**

1. `getToolRegistry(since: Date, agents: AgentSummary[]): Promise<ToolRegistryItem[]>`
   - Calls `fetchToolRegistry()` for tool definitions
   - Queries `agentEvents` for aggregate metrics (call count, failure count, avg latency) grouped by `payload->>'tool_name'` where timestamp >= since
   - Cross-references agent definitions to populate `agents` array per tool
   - Merges registry data with metrics data
   - Returns combined items sorted by namespace then name

2. `getMcpPermissions(): Promise<McpPermission[]>`
   - Queries all three permission tables: `linearMcpPermissions`, `githubMcpPermissions`, `slackMcpPermissions`
   - Maps each row to McpPermission with the namespace derived from which table it came from
   - Returns combined array
   - Uses `db.select().from(table)` pattern for each table, then concatenates results

3. `buildPermissionMatrix(agents: AgentSummary[], mcpPermissions: McpPermission[]): PermissionCell[]`
   - Pure function (no DB access) that cross-references agent YAML tools with MCP permissions
   - Only applies to MCP namespaces: "linear", "github", "slack"
   - For each agent + MCP tool combination: check inYaml (agent.tools includes `namespace:toolName`) and inMcp (permission exists with allowed=true)
   - Also checks MCP permissions not in any YAML (mcp-only mismatches)
   - Returns array of PermissionCell

4. `getToolMetrics(since: Date): Promise<ToolMetrics[]>`
   - Aggregates from `agentEvents` where type IN ('tool.called', 'tool.succeeded', 'tool.failed') AND timestamp >= since
   - Groups by `payload->>'tool_name'`
   - Uses PERCENTILE_CONT for p50 and p95 via `sql` template tag
   - Returns array sorted by callCount DESC
   - Use raw SQL for FILTER clauses and PERCENTILE_CONT (Drizzle doesn't have native helpers)

5. `getToolMetricTimeSeries(since: Date, bucketInterval: string): Promise<ToolMetricTimeSeries[]>`
   - Time-bucketed aggregation for charts
   - `bucketInterval` is a PostgreSQL interval string: '1 minute', '15 minutes', '1 hour'
   - Uses `date_trunc` with the interval for bucketing (but since date_trunc only takes standard values, use `date_trunc('minute', timestamp)` for 1min, or `to_timestamp(floor(extract(epoch from timestamp) / (15*60)) * (15*60))` for 15min, or `date_trunc('hour', timestamp)` for 1h)
   - Actually, simpler approach: use `date_trunc('minute', timestamp)` for 1min buckets, `date_trunc('hour', timestamp)` for 1h buckets. For 15min buckets, use epoch-based floor rounding.
   - Recommended: Accept a `bucketSeconds` number (60, 900, 3600) and use:
     `sql\`to_timestamp(floor(extract(epoch from ${agentEvents.timestamp}) / ${bucketSeconds}) * ${bucketSeconds})\``
   - Change signature to: `getToolMetricTimeSeries(since: Date, bucketSeconds: number): Promise<ToolMetricTimeSeries[]>`
   - Groups by bucket, aggregates calls (tool.called count), failures (tool.failed count), avg latency
   - Returns array sorted by bucket ASC

6. `getRecentToolFailures(params: { limit?: number; offset?: number; namespace?: string; agentId?: string; since?: Date }): Promise<{ items: ToolFailure[]; total: number }>`
   - Queries `agentEvents` where type = 'tool.failed'
   - Applies optional filters: namespace filter by checking `payload->>'tool_name'` starts with tools from that namespace (or join), agentId filter by `agent_definition_id`, since filter by timestamp
   - For namespace filtering: use an approach that checks if tool_name is in the set of tools for that namespace (from registry), OR simpler: just filter by agent_definition_id and timestamp, and let the UI group by namespace. Keep it simple -- namespace filtering can check if the tool_name exists in the registry under that namespace.
   - Actually simplest: just filter by agentId and since. The namespace filter is nice-to-have. If including namespace filter, query the tool registry to get tool names for that namespace and use `inArray`.
   - Returns paginated results with total count (same pattern as listConversations)
   - Maps payload to extract tool_name and output (error message)
   - Order by timestamp DESC
