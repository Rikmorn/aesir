  <files>packages/integrations/src/db/scripts/migrate-tokens.ts</files>
  <action>
Create and run a one-time migration script to move existing token to database:

1. Create packages/integrations/src/db/scripts/migrate-tokens.ts:
   ```typescript
   /**
    * One-Time Token Migration Script
    *
    * Migrates Linear OAuth token from .tokens/linear.json to database.
    * Run with: npx tsx packages/integrations/src/db/scripts/migrate-tokens.ts
    *
    * This script is idempotent - running it multiple times is safe.
    */
   import { readFile } from "node:fs/promises";
   import { storeCredential, getCredentialByProvider } from "../credential-store.js";
   import { createPinoLogger } from "@aesir/common";

   const logger = createPinoLogger({ component: "migration:tokens" });

   const TOKEN_FILE = ".tokens/linear.json";
   const WORKSPACE_ID = "ws_default";

   interface TokenFile {
     accessToken: string;
     refreshToken: string;
     expiresAt: number;
   }

   async function migrate() {
     logger.info("Starting token migration...");

     // Check if already migrated
     const existing = await getCredentialByProvider(WORKSPACE_ID, "linear");
     if (existing) {
       logger.info(
         { credentialId: existing.id },
         "Linear credential already exists in database, skipping migration"
       );
       return;
     }

     // Read token file
     let tokenData: TokenFile;
     try {
       const content = await readFile(TOKEN_FILE, "utf-8");
       tokenData = JSON.parse(content) as TokenFile;
       logger.info("Read token file successfully");
     } catch (err) {
       if (err instanceof Error && "code" in err && err.code === "ENOENT") {
         logger.info("No token file found, nothing to migrate");
         return;
       }
       throw err;
     }

     // Validate token data
     if (!tokenData.accessToken) {
       logger.error("Token file missing accessToken, cannot migrate");
       return;
     }

     // Store in database
     const credentialId = await storeCredential({
       workspaceId: WORKSPACE_ID,
       provider: "linear",
       accessToken: tokenData.accessToken,
       refreshToken: tokenData.refreshToken || undefined,
       expiresAt: tokenData.expiresAt ? new Date(tokenData.expiresAt) : undefined,
     });

     logger.info({ credentialId }, "Linear token migrated to database successfully");
   }

   migrate()
     .then(() => {
       logger.info("Migration complete");
       process.exit(0);
     })
     .catch((err) => {
       logger.error({ err }, "Migration failed");
       process.exit(1);
     });
   ```

2. Ensure CREDENTIAL_ENCRYPTION_KEY is set in environment:

   ```bash
   export CREDENTIAL_ENCRYPTION_KEY=$(openssl rand -hex 32)

   ```
