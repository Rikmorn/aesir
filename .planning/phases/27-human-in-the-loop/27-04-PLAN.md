   ```typescript
   export async function sendApprovalSignal(
     deps: SignalHandlerDeps,
     input: {
       taskIdentifier: string;  // e.g., "ABC-123"
       taskId?: string;         // UUID if known
       approved: boolean;
       feedback?: string;
       approverUserId?: string;
       channel?: string;
     }
   ): Promise<{ signaled: boolean; workflowId?: string; error?: string }> {
     const { workflowClient, logger } = deps;

     // Workflow ID format: dev-agent-{issueUUID}
     // Problem: We have identifier (ABC-123) but need UUID
     // Solution: Query existing workflows or use identifier-based ID

     // Try workflow ID patterns:
     // 1. If taskId provided, use it directly
     // 2. Otherwise, try to find workflow by identifier (future: workflow query)

     const workflowId = input.taskId
       ? `dev-agent-${input.taskId}`
       : `dev-agent-${input.taskIdentifier}`; // Fallback - may need adjustment

     try {
       const handle = workflowClient.workflow.getHandle(workflowId);

       await handle.signal(planApprovalSignal, {
         approved: input.approved,
         feedback: input.feedback,
       });

       logger.info({ workflowId, approved: input.approved }, "Approval signal sent");
       return { signaled: true, workflowId };
     } catch (error) {
       // Check if workflow not found
       const isNotFound = error instanceof Error &&
         (error.message.includes("not found") ||
          error.name === "WorkflowNotFoundError");

       if (isNotFound) {
         logger.warn({ workflowId }, "Workflow not found for approval signal");
         return { signaled: false, error: "Workflow not found" };
       }

       logger.error({ err: error, workflowId }, "Failed to send approval signal");
       throw error;
     }
   }
   ```

   ```typescript
     const { logger } = deps;

     // Extract task identifier from branch name
     // Branch format: feature/{identifier} (e.g., feature/ABC-123)

     if (!branchMatch) {
       logger.warn({ branchName: input.branchName }, "Cannot extract task ID from branch");
       return { signaled: false, error: "Cannot identify task from branch name" };
     }

     logger.info({ taskIdentifier, merged: input.merged }, "Processing PR completion");

     // For now, just log - actual completion handling in plan 06
   ```
