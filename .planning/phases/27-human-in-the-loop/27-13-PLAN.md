   ```typescript
   export const CommentPayloadSchema = z.object({
     action: z.enum(["create", "update", "remove"]),
     type: z.literal("Comment"),
     data: z.object({
       id: z.string(),
       body: z.string(),
       issueId: z.string(),
       userId: z.string(),
       createdAt: z.string(),
     }),
     // User who authored the comment
     actor: z.object({
       id: z.string(),
       name: z.string(),
       email: z.string().optional(),
     }).optional(),
   });

   export type CommentPayload = z.infer<typeof CommentPayloadSchema>;
   ```

2. Add parse function:
   ```typescript
   export function parseCommentPayload(body: unknown): CommentPayload | null {
     const result = CommentPayloadSchema.safeParse(body);
     if (!result.success) {
       return null;
     }
     return result.data;
   }
   ```

Export the schema and function.
