3. Update sendCompletionSignal to actually send the signal:
   ```typescript
   export async function sendCompletionSignal(
     deps: SignalHandlerDeps,
     input: {
       prNumber: number;
       merged: boolean;
       branchName: string;
       repository: { owner: string; name: string };
     }
   ): Promise<{ signaled: boolean; workflowId?: string; error?: string }> {
     const { workflowClient, logger } = deps;

     // Extract task identifier from branch name
     const branchMatch = input.branchName.match(/feature\/([A-Z]+-\d+)/i);
     if (!branchMatch) {
       logger.warn({ branchName: input.branchName }, "Cannot extract task ID from branch");
       return { signaled: false, error: "Cannot identify task from branch name" };
     }

     const taskIdentifier = branchMatch[1];
     const workflowId = `dev-agent-${taskIdentifier}`;

     logger.info(
       { taskIdentifier, merged: input.merged, workflowId },
       "Sending PR completion signal"
     );

     try {
       const handle = workflowClient.workflow.getHandle(workflowId);

       const payload: PRCompletionPayload = {
         merged: input.merged,
         prNumber: input.prNumber,
         branchName: input.branchName,
       };

       await handle.signal(prCompletionSignal, payload);

       logger.info({ workflowId, merged: input.merged }, "PR completion signal sent");
       return { signaled: true, workflowId };
     } catch (error) {
       const isNotFound = error instanceof Error &&
         (error.message.includes("not found") ||
          error.name === "WorkflowNotFoundError");

       if (isNotFound) {
         logger.warn({ workflowId }, "Workflow not found for PR completion signal");
         return { signaled: false, error: "Workflow not found or already completed" };
       }

       logger.error({ err: error, workflowId }, "Failed to send PR completion signal");
       throw error;
     }
   }
   ```
