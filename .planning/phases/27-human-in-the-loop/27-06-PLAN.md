        });

      ```typescript

      await callMcpTool({
        integration: "slack",
        tool: "send_message",
        params: {
          channel: deps.slackChannel,
          text: completionMessage,
          blocks: buildCompletionBlocks(state, stats),
        },
        agentId: "dev-agent",
        correlationId: `complete-${state.taskId}`,
      });
      ```

   d. Cleanup container:
      ```typescript
      if (state.containerId) {
        try {
          await deps.manager.cleanup(state.containerId);
          nodeLogger.info({ containerId: state.containerId }, "Container cleaned up");
        } catch (error) {
          // Non-critical - cleanup service will catch orphans
          nodeLogger.warn({ err: error }, "Failed to cleanup container");
        }
      }
      ```

   e. Return final state:
      ```typescript
      return {
        phase: "complete",
      };
      ```

4. Helper functions:

   buildCompletionStats:
   ```typescript
   interface CompletionStats {
     filesChanged: number;
     linesAdded: number;
     linesRemoved: number;
     executionSteps: number;
     timeToComplete?: string;  // Calculated if start time available
   }

   function buildCompletionStats(state: DevAgentState): CompletionStats {
     const filesChanged = state.files?.length ?? 0;
     const linesAdded = state.files?.reduce((sum, f) =>
       sum + (f.content?.split('\n').length ?? 0), 0) ?? 0;

     return {
       filesChanged,
       linesAdded,
       linesRemoved: 0,  // Would need git diff for accurate count
       executionSteps: state.executionPlan?.steps?.length ?? 0,
     };
   }
   ```

   formatCompletionMessage:
   ```typescript
   function formatCompletionMessage(state: DevAgentState, stats: CompletionStats): string {
     return `:white_check_mark: *${state.issue?.identifier} completed!*\n\n` +
       `*${state.issue?.title}*\n\n` +
       `PR: ${state.prUrl}\n` +
       `Files: ${stats.filesChanged} | Steps: ${stats.executionSteps}`;
   }
   ```

   buildCompletionBlocks (Block Kit):
   ```typescript
   function buildCompletionBlocks(state: DevAgentState, stats: CompletionStats) {
     return [
       {
         type: "section",
         text: {
           type: "mrkdwn",
           text: `:white_check_mark: *${state.issue?.identifier} completed!*`,
         },
       },
       {
         type: "section",
         fields: [
           { type: "mrkdwn", text: `*Title:*\n${state.issue?.title}` },
           { type: "mrkdwn", text: `*PR:*\n<${state.prUrl}|#${state.prNumber}>` },
         ],
       },
       {
         type: "context",
         elements: [
           {
             type: "mrkdwn",
             text: `${stats.filesChanged} files | ${stats.executionSteps} steps`,
           },
         ],
       },
     ];
   }
   ```
