/**
 * Zod schemas for GitHub MCP tools
 *
 * Defines input and output schemas for all GitHub MCP tools.

import { z } from "zod";

// === REPOSITORY TOOLS ===

export const GetRepositoryInputSchema = z.object({
  owner: z.string().describe("Repository owner (user or organization)"),
  repo: z.string().describe("Repository name"),
});

export const RepositoryOutputSchema = z.object({
  id: z.number(),
  name: z.string(),
  fullName: z.string(),
  description: z.string().optional(),
  defaultBranch: z.string(),
  private: z.boolean(),
  url: z.string().url(),
});

// === BRANCH TOOLS ===

export const CreateBranchInputSchema = z.object({
  owner: z.string().describe("Repository owner"),
  repo: z.string().describe("Repository name"),
  branchName: z.string().describe("Name for the new branch"),
  sourceBranch: z.string().default("main").describe("Branch to create from (default: main)"),
});

export const BranchOutputSchema = z.object({
  name: z.string(),
  sha: z.string(),
  url: z.string().url(),
});

// === COMMIT TOOLS ===

export const CreateCommitInputSchema = z.object({
  owner: z.string().describe("Repository owner"),
  repo: z.string().describe("Repository name"),
  branch: z.string().describe("Branch to commit to"),
  message: z.string().describe("Commit message"),
  files: z.array(z.object({
    path: z.string().describe("File path relative to repository root"),
    content: z.string().describe("File content (text)"),
  })).describe("Files to create or update"),
});

export const CommitOutputSchema = z.object({
  sha: z.string(),
  message: z.string(),
  url: z.string().url(),
});

// === PULL REQUEST TOOLS ===

export const CreatePRInputSchema = z.object({
  owner: z.string().describe("Repository owner"),
  repo: z.string().describe("Repository name"),
  title: z.string().min(1).describe("PR title"),
  body: z.string().optional().describe("PR description in markdown"),
  head: z.string().describe("Head branch name (source branch with changes)"),
  base: z.string().default("main").describe("Base branch name (target branch)"),
});

export const PROutputSchema = z.object({
  number: z.number(),
  title: z.string(),
  state: z.enum(["open", "closed"]),
  merged: z.boolean(),
  url: z.string().url(),
});

export const GetPRInputSchema = z.object({
  owner: z.string().describe("Repository owner"),
  repo: z.string().describe("Repository name"),
  pullNumber: z.number().describe("Pull request number"),
});

export const ListPRsInputSchema = z.object({
  owner: z.string().describe("Repository owner"),
  repo: z.string().describe("Repository name"),
  state: z.enum(["open", "closed", "all"]).default("open").describe("Filter by PR state"),
  perPage: z.number().min(1).max(100).default(30).describe("Results per page"),
});

export const ListPRsOutputSchema = z.object({
  pullRequests: z.array(PROutputSchema),
  total: z.number(),
});

export const MergePRInputSchema = z.object({
  owner: z.string().describe("Repository owner"),
  repo: z.string().describe("Repository name"),
  pullNumber: z.number().describe("Pull request number"),
  mergeMethod: z.enum(["merge", "squash", "rebase"]).default("squash").describe("Merge method"),
});

export const MergePROutputSchema = z.object({
  merged: z.boolean(),
  sha: z.string().optional(),
  message: z.string(),
});

// === FILE TOOLS ===

export const GetFileContentsInputSchema = z.object({
  owner: z.string().describe("Repository owner"),
  repo: z.string().describe("Repository name"),
  path: z.string().describe("File path relative to repository root"),
  ref: z.string().optional().describe("Branch, tag, or commit SHA (default: default branch)"),
});

export const FileContentsOutputSchema = z.object({
  path: z.string(),
  content: z.string(),
  sha: z.string(),
  size: z.number(),
  encoding: z.string(),
});

export const ListFilesInputSchema = z.object({
  owner: z.string().describe("Repository owner"),
  repo: z.string().describe("Repository name"),
  path: z.string().default("").describe("Directory path (default: root)"),
  ref: z.string().optional().describe("Branch, tag, or commit SHA"),

export const FileEntrySchema = z.object({
  name: z.string(),
  path: z.string(),
  type: z.enum(["file", "dir", "submodule", "symlink"]),
  size: z.number().optional(),
});

export const ListFilesOutputSchema = z.object({
  files: z.array(FileEntrySchema),
});

// Type exports
export type GetRepositoryInput = z.infer<typeof GetRepositoryInputSchema>;
export type RepositoryOutput = z.infer<typeof RepositoryOutputSchema>;
export type CreateBranchInput = z.infer<typeof CreateBranchInputSchema>;
export type BranchOutput = z.infer<typeof BranchOutputSchema>;
export type CreateCommitInput = z.infer<typeof CreateCommitInputSchema>;
export type CommitOutput = z.infer<typeof CommitOutputSchema>;
export type CreatePRInput = z.infer<typeof CreatePRInputSchema>;
export type PROutput = z.infer<typeof PROutputSchema>;
export type GetPRInput = z.infer<typeof GetPRInputSchema>;
export type ListPRsInput = z.infer<typeof ListPRsInputSchema>;
export type ListPRsOutput = z.infer<typeof ListPRsOutputSchema>;
export type MergePRInput = z.infer<typeof MergePRInputSchema>;
export type MergePROutput = z.infer<typeof MergePROutputSchema>;
export type GetFileContentsInput = z.infer<typeof GetFileContentsInputSchema>;
export type FileContentsOutput = z.infer<typeof FileContentsOutputSchema>;
export type ListFilesInput = z.infer<typeof ListFilesInputSchema>;

**mcp/server.ts** - MCP server factory:
```typescript
/**
 * GitHub MCP Server Factory
 *
 * Creates an MCP server instance with all GitHub tools registered.
 */

import { Server } from "@modelcontextprotocol/sdk/server/index.js";
import type { PostgresJsDatabase } from "drizzle-orm/postgres-js";
import type { Logger } from "pino";
import { registerRepositoryTools } from "./tools/repository.js";
import { registerPRTools } from "./tools/pullrequests.js";
import { registerFileTools } from "./tools/files.js";

export interface CreateGitHubMCPServerOptions {
  db: PostgresJsDatabase;
  logger: Logger;
  owner?: string; // Default owner for credential lookup
}

export function createGitHubMCPServer(options: CreateGitHubMCPServerOptions): Server {
  const { db, logger, owner = "default" } = options;

  const childLogger = logger.child({ component: "integrations:github:mcp" });

  childLogger.info("Creating GitHub MCP server");

  const server = new Server(
    {
      name: "github-mcp-server",
      version: "1.0.0",
    },
    {
      capabilities: {
        tools: {
          listChanged: true,
        },
      },
    }
  );

  // Register all tool handlers
  const toolDeps = { server, db, logger: childLogger, owner };

  registerRepositoryTools(toolDeps);
  registerPRTools(toolDeps);
  registerFileTools(toolDeps);

  childLogger.info("GitHub MCP server created with tools registered");

  return server;
}
```
