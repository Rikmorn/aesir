/**
 * Zod schemas for Linear MCP tools
 *
 * Defines input and output schemas for all Linear MCP tools.
 * Schemas are used for:
 * 1. Runtime validation of tool inputs
 * 2. JSON Schema generation for MCP tool registration
 * 3. Type inference for TypeScript
 */

import { z } from "zod";

// === ISSUE TOOLS ===

export const GetIssueInputSchema = z.object({
  issueId: z.string().describe("Linear issue ID or identifier (e.g., 'ABC-123' or UUID)"),
});

export const IssueOutputSchema = z.object({
  id: z.string(),
  identifier: z.string().describe("Human-readable issue identifier (e.g., 'ABC-123')"),
  title: z.string(),
  description: z.string().optional(),
  status: z.string(),
  priority: z.number().optional(),
  assignee: z.string().optional().describe("Assignee name, if assigned"),
  url: z.string().url(),
});

export const CreateIssueInputSchema = z.object({
  teamId: z.string().describe("Linear team ID (get from list_teams)"),
  title: z.string().min(1).describe("Issue title"),
  description: z.string().optional().describe("Issue description in markdown"),
  priority: z.number().min(0).max(4).optional().describe("Priority: 0=no priority, 1=urgent, 2=high, 3=medium, 4=low"),
  labelIds: z.array(z.string()).optional().describe("Array of label IDs (get from list_labels)"),
});

export const CreateIssueOutputSchema = z.object({
  id: z.string(),
  identifier: z.string(),
  url: z.string().url(),
  status: z.string(),
});

export const UpdateIssueStatusInputSchema = z.object({
  issueId: z.string().describe("Linear issue ID or identifier"),
  stateId: z.string().describe("Target workflow state ID (get from issue's team's workflow states)"),
});

export const UpdateIssueStatusOutputSchema = z.object({
  id: z.string(),
  identifier: z.string(),
  status: z.string(),
  url: z.string().url(),
});

// === TEAM TOOLS ===

export const ListTeamsInputSchema = z.object({}).describe("No input required");

export const TeamSchema = z.object({
  id: z.string(),
  name: z.string(),
  key: z.string().describe("Team key prefix used in issue identifiers"),
});

export const ListTeamsOutputSchema = z.object({
  teams: z.array(TeamSchema),
});

// === LABEL TOOLS ===

export const ListLabelsInputSchema = z.object({
  teamId: z.string().optional().describe("Filter labels by team ID (optional)"),
});

export const LabelSchema = z.object({
  id: z.string(),
  name: z.string(),
  color: z.string().optional(),
});

export const ListLabelsOutputSchema = z.object({
  labels: z.array(LabelSchema),
});

export type IssueOutput = z.infer<typeof IssueOutputSchema>;
export type CreateIssueInput = z.infer<typeof CreateIssueInputSchema>;
export type CreateIssueOutput = z.infer<typeof CreateIssueOutputSchema>;
export type UpdateIssueStatusInput = z.infer<typeof UpdateIssueStatusInputSchema>;
export type UpdateIssueStatusOutput = z.infer<typeof UpdateIssueStatusOutputSchema>;
export type ListTeamsOutput = z.infer<typeof ListTeamsOutputSchema>;
export type ListLabelsInput = z.infer<typeof ListLabelsInputSchema>;
export type ListLabelsOutput = z.infer<typeof ListLabelsOutputSchema>;

/**
 * Linear MCP Tools - Issues
 *
 * Provides MCP tools for Linear issue operations:
 * - get_issue: Retrieve issue details
 * - create_issue: Create new issue
 * - update_issue_status: Change issue workflow state
 */

import type { Server } from "@modelcontextprotocol/sdk/server/index.js";
import type { PostgresJsDatabase } from "drizzle-orm/postgres-js";
import type { Logger } from "pino";
import { zodToJsonSchema } from "zod-to-json-schema";
import { createToolResult, createErrorResult, type MCPToolContext } from "@aesir/common";
import { checkLinearToolPermission } from "../../db/permissions.js";
import { createLinearClientFromDatabase } from "../../oauth/client.js";
import { readIssue, createIssue, updateIssue } from "../../client/issues.js";
import {
  GetIssueInputSchema,
  CreateIssueInputSchema,
  UpdateIssueStatusInputSchema,
  type IssueOutput,
  type CreateIssueOutput,
  type UpdateIssueStatusOutput,
} from "../schemas.js";

interface RegisterToolsDeps {
  server: Server;
  db: PostgresJsDatabase;
  logger: Logger;
  workspaceId: string;
}

/**
 * Register issue-related MCP tools
 */
export function registerIssueTools(deps: RegisterToolsDeps): void {
  const { server, db, logger, workspaceId } = deps;

  // get_issue
  server.setRequestHandler("tools/call", async (request) => {
    if (request.params.name !== "get_issue") return null;

    const context: MCPToolContext = {
      logger: logger.child({ tool: "get_issue" }),
      correlationId: (request.params.arguments as Record<string, unknown>)?._correlationId as string || "unknown",
      agentId: (request.params.arguments as Record<string, unknown>)?._agentId as string || "unknown",
      startTime: Date.now(),
    };

    // Permission check
    const permitted = await checkLinearToolPermission(
      { db, logger: context.logger },
      { agentId: context.agentId, toolName: "get_issue" }
    );

    if (!permitted) {
      context.logger.warn("Permission denied");
      return createErrorResult(context, "Permission denied: get_issue not allowed for this agent");
    }

    // Validate input
    const parseResult = GetIssueInputSchema.safeParse(request.params.arguments);
    if (!parseResult.success) {
      return createErrorResult(context, `Invalid input: ${parseResult.error.message}`);
    }

    const { issueId } = parseResult.data;
    context.logger.info({ issueId }, "Fetching Linear issue");

    try {
      const client = await createLinearClientFromDatabase({ workspaceId, db, logger: context.logger });
      const issue = await readIssue(client, issueId);

      if (!issue) {
        return createErrorResult(context, `Issue not found: ${issueId}`);
      }

      const assignee = issue.assignee ? await issue.assignee : null;
      const state = issue.state ? await issue.state : null;

      const output: IssueOutput = {
        id: issue.id,
        identifier: issue.identifier,
        title: issue.title,
        description: issue.description || undefined,
        status: state?.name || "Unknown",
        priority: issue.priority,
        assignee: assignee?.name,
        url: issue.url,
      };

      context.logger.info({ issueId: issue.id }, "Issue fetched successfully");
      return createToolResult(context, `Issue ${issue.identifier}: ${issue.title}`, output);
    } catch (error) {
      context.logger.error({ err: error }, "Failed to fetch issue");
      return createErrorResult(context, `Failed to fetch issue: ${(error as Error).message}`);
    }
  });

  // Register tool metadata
  server.setRequestHandler("tools/list", async () => {
    return {
      tools: [
        {
          name: "get_issue",
          description: "Retrieve details about a Linear issue. Use this when you need to check issue status, read description, or see who is assigned.",
          inputSchema: zodToJsonSchema(GetIssueInputSchema),
        },
        {
          name: "create_issue",
          description: "Create a new Linear issue in a team. Use this when you need to create a task, bug report, or feature request. Get teamId from list_teams first.",
          inputSchema: zodToJsonSchema(CreateIssueInputSchema),
        },
        {
          name: "update_issue_status",
          description: "Update the workflow status of a Linear issue. Use this to move an issue through its workflow (e.g., To Do -> In Progress -> Done).",
          inputSchema: zodToJsonSchema(UpdateIssueStatusInputSchema),
        },
      ],
    };
  });
}
```

/**
 * Linear MCP Server Factory
 *
 * Creates an MCP server instance with all Linear tools registered.
 * The server can be mounted on Express routes for HTTP/SSE transport.
 */

import { Server } from "@modelcontextprotocol/sdk/server/index.js";
import type { PostgresJsDatabase } from "drizzle-orm/postgres-js";
import type { Logger } from "pino";
import { registerIssueTools } from "./tools/issues.js";
import { registerTeamTools } from "./tools/teams.js";

export interface CreateLinearMCPServerOptions {
  /** Database connection for permission checks and credential lookup */
  db: PostgresJsDatabase;
  /** Pino logger for structured logging */
  logger: Logger;
  /** Workspace ID for credential lookup (default: "ws_default") */
  workspaceId?: string;
}

/**
 * Create a Linear MCP server with all tools registered
 *
 * @returns MCP Server instance ready for transport binding
 */
export function createLinearMCPServer(options: CreateLinearMCPServerOptions): Server {
  const { db, logger, workspaceId = "ws_default" } = options;

  const childLogger = logger.child({ component: "integrations:linear:mcp" });

  childLogger.info("Creating Linear MCP server");

  const server = new Server(
    {
      name: "linear-mcp-server",
      version: "1.0.0",
    },
    {
      capabilities: {
        tools: {
          listChanged: true, // Notify clients when tools change
        },
      },
    }
  );

  // Register all tool handlers
  const toolDeps = { server, db, logger: childLogger, workspaceId };

  registerIssueTools(toolDeps);
  registerTeamTools(toolDeps);

  childLogger.info("Linear MCP server created with tools registered");

  return server;
}
```
