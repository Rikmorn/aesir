<task type="auto">
  <name>Task 2: Implement writeFile and readFile</name>
  <files>src/sandbox/docker-sandbox.ts</files>
  <action>
    Implement the stubbed writeFile() and readFile() methods in DockerSandbox.

    **writeFile(path, content):**
    - Create a tar archive containing the file
    - Use container.putArchive() to write to container
    - Path should be absolute in container (e.g., /app/index.ts)
    - Use tar-fs or tar-stream to create archive from content string

    Pattern from RESEARCH.md:
    ```typescript
    import tar from 'tar-fs'
    import { Readable } from 'stream'

    // Create tar with single file
    const pack = tar.pack(/* base dir */, {
      entries: [filename],
      map: (header) => { /* set path */ }
    })
    await container.putArchive(pack, { path: dirname })
    ```

    **Alternative simpler approach** - create in-memory tar:
    ```typescript
    import { pack } from 'tar-stream'

    async writeFile(filePath: string, content: string): Promise<void> {
      const pack = tar.pack()
      const filename = path.basename(filePath)
      const dirname = path.dirname(filePath)

      pack.entry({ name: filename }, content)
      pack.finalize()

      await this.container.putArchive(pack, { path: dirname })
    }
    ```

    **readFile(path):**
    - Use container.getArchive() which returns tar stream
    - Extract the file content from the tar
    - Return as string

    Pattern:
    ```typescript
    async readFile(filePath: string): Promise<string> {
      const stream = await this.container.getArchive({ path: filePath })
      // Extract file from tar stream
      return extractFileFromTar(stream)
    }
    ```

    Use tar-stream's extract() to parse the archive.

    **Error handling:**
    - Throw descriptive errors if file doesn't exist
    - Log operations using Logger

    **Note:** tar-stream may be simpler than tar-fs for single-file operations. Either library is acceptable per RESEARCH.md.
  </action>

  <name>Task 3: Implement runTests</name>
  <files>src/sandbox/docker-sandbox.ts</files>
  <action>
    Implement runTests() method that wraps execute() with test-specific parsing.

    ```typescript
    async runTests(command: string[]): Promise<TestResult> {
      const result = await this.execute(command)

      return {
        ...result,
        passed: result.exitCode === 0,
        summary: result.exitCode === 0
          ? 'All tests passed'
          : `Tests failed with exit code ${result.exitCode}`
      }
    }
    ```

    This is intentionally simple for MVP:
    - exitCode === 0 means tests passed
    - exitCode !== 0 means tests failed
    - Summary is human-readable

    Future enhancement could parse JSON test output, but that's not in scope for MVP.
  </action>

    **Test structure:**
    ```typescript
    import { describe, it, expect, beforeAll, afterAll } from 'vitest'
    import { DockerSandbox } from './docker-sandbox'
    import type { Sandbox } from './types'

    describe('DockerSandbox', () => {
      // Use longer timeout for Docker operations
      // Tests require Docker daemon running

      describe('execute', () => {
        it('should execute simple command and capture stdout', async () => {
          const sandbox = await DockerSandbox.create()
          try {
            const result = await sandbox.execute(['echo', 'hello world'])
            expect(result.exitCode).toBe(0)
            expect(result.stdout).toContain('hello world')
            expect(result.stderr).toBe('')
          } finally {
            await sandbox.cleanup()
          }
        })

        it('should capture stderr for failed commands', async () => {
          const sandbox = await DockerSandbox.create()
          try {
            const result = await sandbox.execute(['ls', '/nonexistent'])
            expect(result.exitCode).not.toBe(0)
            expect(result.stderr).toContain('No such file')
          } finally {
            await sandbox.cleanup()
          }
        })

        it('should return correct exit code', async () => {
          const sandbox = await DockerSandbox.create()
          try {
            const result = await sandbox.execute(['sh', '-c', 'exit 42'])
            expect(result.exitCode).toBe(42)
          } finally {
            await sandbox.cleanup()
          }
        })
      })

      describe('file operations', () => {
        it('should write and read files', async () => {
          const sandbox = await DockerSandbox.create()
          try {
            await sandbox.writeFile('/tmp/test.txt', 'hello sandbox')
            const content = await sandbox.readFile('/tmp/test.txt')
            expect(content).toBe('hello sandbox')
          } finally {
            await sandbox.cleanup()
          }
        })

        it('should handle multi-line content', async () => {
          const sandbox = await DockerSandbox.create()
          try {
            const content = 'line1\nline2\nline3'
            await sandbox.writeFile('/tmp/multi.txt', content)
            const result = await sandbox.readFile('/tmp/multi.txt')
            expect(result).toBe(content)
          } finally {
            await sandbox.cleanup()
          }
        })
      })

      describe('runTests', () => {
        it('should return passed=true for exit code 0', async () => {
          const sandbox = await DockerSandbox.create()
          try {
            const result = await sandbox.runTests(['sh', '-c', 'exit 0'])
            expect(result.passed).toBe(true)
            expect(result.exitCode).toBe(0)
          } finally {
            await sandbox.cleanup()
          }
        })

        it('should return passed=false for non-zero exit', async () => {
          const sandbox = await DockerSandbox.create()
          try {
            const result = await sandbox.runTests(['sh', '-c', 'exit 1'])
            expect(result.passed).toBe(false)
            expect(result.exitCode).toBe(1)
          } finally {
            await sandbox.cleanup()
          }
        })
      })

          const containerId = sandbox.containerId // Add getter if needed

          // Could verify with docker.getContainer(id).inspect() throws
        })
      })
    })
    ```
