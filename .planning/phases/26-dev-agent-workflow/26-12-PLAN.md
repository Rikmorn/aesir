  <action>
Add or update dev-agent service in docker-compose.yml:

```yaml
  dev-agent:
    build:
      context: .
      dockerfile: packages/agents/Dockerfile
    command: ["node", "dist/dev-agent/main.js"]
    environment:
      - NODE_ENV=development
      - DEV_AGENT_PORT=3004
      - DEV_AGENT_SLACK_CHANNEL=${DEV_AGENT_SLACK_CHANNEL}
      - TEMPORAL_ADDRESS=temporal:7233
      - DATABASE_URL=postgresql://postgres:postgres@postgresql:5432/aesir
      - ANTHROPIC_API_KEY=${ANTHROPIC_API_KEY}
      - GITHUB_REPO_URL=${GITHUB_REPO_URL}
      - GITHUB_TOKEN=${GITHUB_TOKEN}
      - GITHUB_OWNER=${GITHUB_OWNER}
      - GITHUB_REPO=${GITHUB_REPO}
      - GITHUB_BASE_BRANCH=${GITHUB_BASE_BRANCH:-main}
      - LINEAR_MCP_URL=http://linear-integration:3001
      - GITHUB_MCP_URL=http://github-integration:3002
      - SLACK_MCP_URL=http://slack-integration:3003
    ports:
      - "3004:3004"
    volumes:
      # Mount Docker socket for DevContainerManager
      - /var/run/docker.sock:/var/run/docker.sock
      # macOS alternative path
      - ${HOME}/.docker/run/docker.sock:/var/run/docker.sock:ro
    depends_on:
      postgresql:
        condition: service_healthy
      temporal:
        condition: service_healthy
      linear-integration:
        condition: service_started
      github-integration:
        condition: service_started
      slack-integration:
        condition: service_started
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:3004/health"]
      interval: 10s
      timeout: 5s
      retries: 3
    restart: unless-stopped

  # Dev-agent worker (separate service for Temporal)
  dev-agent-worker:
    build:
      context: .
      dockerfile: packages/agents/Dockerfile
    command: ["node", "dist/dev-agent/worker.js"]
    environment:
      - NODE_ENV=development
      - TEMPORAL_ADDRESS=temporal:7233
      - DATABASE_URL=postgresql://postgres:postgres@postgresql:5432/aesir
      - ANTHROPIC_API_KEY=${ANTHROPIC_API_KEY}
      - GITHUB_REPO_URL=${GITHUB_REPO_URL}
      - GITHUB_TOKEN=${GITHUB_TOKEN}
      - GITHUB_OWNER=${GITHUB_OWNER}
      - GITHUB_REPO=${GITHUB_REPO}
      - GITHUB_BASE_BRANCH=${GITHUB_BASE_BRANCH:-main}
      - DEV_AGENT_SLACK_CHANNEL=${DEV_AGENT_SLACK_CHANNEL}
      - LINEAR_MCP_URL=http://linear-integration:3001
      - GITHUB_MCP_URL=http://github-integration:3002
      - SLACK_MCP_URL=http://slack-integration:3003
    volumes:
      - /var/run/docker.sock:/var/run/docker.sock
      - ${HOME}/.docker/run/docker.sock:/var/run/docker.sock:ro
    depends_on:
      postgresql:
        condition: service_healthy
      temporal:
        condition: service_healthy
      dev-agent:
        condition: service_healthy
    restart: unless-stopped
```

Notes:
- Docker socket mount enables DevContainerManager to spawn containers
- Both Linux (/var/run/docker.sock) and macOS paths included
- Worker runs as separate service for Temporal task processing
- Depends on all integration services for MCP calls

<task type="auto">
  <name>Task 3: Add nginx routing for dev-agent</name>
  <files>nginx/nginx.conf</files>
  <action>
If nginx config exists, add routing for dev-agent.

Add location block:

```nginx
# Dev Agent (port 3004)
location /agent/ {
    proxy_pass http://dev-agent:3004/;
    proxy_http_version 1.1;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
}
```

This allows external access if needed (e.g., for testing). Internal Docker network access works without nginx.

If nginx.conf doesn't exist or isn't used for dev-agent routing, skip this task.
