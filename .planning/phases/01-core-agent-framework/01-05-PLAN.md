### Task 1: Mock LLM for Testing

**What:** Create a mock LLM that simulates Claude responses for deterministic testing.

**File to create:** `src/testing/mock-llm.ts`

**Implementation:**
```typescript
import { BaseChatModel } from "@langchain/core/language_models/chat_models";
import { AIMessage, BaseMessage } from "@langchain/core/messages";
import { ChatResult } from "@langchain/core/outputs";

export class MockChatModel extends BaseChatModel {
  private responses: string[];
  private callCount = 0;
  private shouldLoop: boolean;

  constructor(options: { responses?: string[]; shouldLoop?: boolean } = {}) {
    super({});
    this.responses = options.responses ?? ["I'll generate some code for you."];
    this.shouldLoop = options.shouldLoop ?? false;
  }

  _llmType(): string {
    return "mock";
  }

  async _generate(
    messages: BaseMessage[],
  ): Promise<ChatResult> {
    this.callCount++;

    // If shouldLoop, always return a tool call to simulate infinite loop
    if (this.shouldLoop) {
      return {
        generations: [{
          text: "",
          message: new AIMessage({
            content: "",
            tool_calls: [{
              name: "generate_code",
              args: { taskDescription: "loop", language: "typescript" },
              id: `call_${this.callCount}`,
            }],
          }),
        }],
      };
    }

    const responseIndex = Math.min(this.callCount - 1, this.responses.length - 1);
    return {
      generations: [{
        text: this.responses[responseIndex],
        message: new AIMessage({
          content: this.responses[responseIndex],
        }),
      }],
    };
  }

  getCallCount(): number {
    return this.callCount;
  }
}
```

      expect(result.success).toBe(false);
      expect(result.terminationReason).toBe("recursion_limit");
    });

    it("should track loop count in state", async () => {
      // Verify loop counter increments
    });
  });

  describe("CORE-03: Wall-clock timeout", () => {
    it("should stop execution when timeout is exceeded", async () => {
      // Use mock LLM with artificial delay
      const result = await runAgentWithGuardrails(
        "Generate code",
        "test-thread-3",
        { timeoutMs: 100 } // Very short timeout
      );

      expect(result.success).toBe(false);
      expect(result.terminationReason).toBe("timeout");
    });
  });

  describe("CORE-04: Activity logging", () => {
    it("should log agent_start with timestamp and context", async () => {
      // Capture logs during execution
      const logs = captureLogsForThread("test-thread-4");

      await runAgentWithGuardrails("Generate code", "test-thread-4");

      const startLog = logs.find(l => l.action === "agent_start");
      expect(startLog).toBeDefined();
      expect(startLog?.timestamp).toMatch(/^\d{4}-\d{2}-\d{2}T/); // ISO 8601
      expect(startLog?.context.threadId).toBe("test-thread-4");
    });

    it("should log agent_complete with outcome and duration", async () => {
      const logs = captureLogsForThread("test-thread-5");

      await runAgentWithGuardrails("Generate code", "test-thread-5");

      const completeLog = logs.find(l => l.action === "agent_complete");
      expect(completeLog?.outcome).toBe("success");
      expect(completeLog?.durationMs).toBeGreaterThan(0);
    });
  });

  describe("CORE-05: Agents defined via code/config", () => {
    it("should have valid agent configuration schema", () => {
      const result = AgentConfigSchema.safeParse(devAgentConfig);
      expect(result.success).toBe(true);
    });

    it("should reject invalid configuration", () => {
      const invalid = { ...devAgentConfig, maxIterations: -1 };
      const result = AgentConfigSchema.safeParse(invalid);
      expect(result.success).toBe(false);
    });

    it("should have langgraph.json with graph definition", async () => {
      const fs = await import("fs/promises");
      const config = JSON.parse(await fs.readFile("langgraph.json", "utf-8"));

      expect(config.graphs).toBeDefined();
      expect(config.graphs.dev_agent).toBeDefined();
    });
  });

import { LogEntry } from "../logging";

let capturedLogs: LogEntry[] = [];

export function startLogCapture(): void {
  capturedLogs = [];
  // Intercept logger output
}

export function getCapturedLogs(): LogEntry[] {
  return [...capturedLogs];
}

export function getLogsForThread(threadId: string): LogEntry[] {
  return capturedLogs.filter(log => log.context.threadId === threadId);
}

export function clearCapturedLogs(): void {
  capturedLogs = [];
}
