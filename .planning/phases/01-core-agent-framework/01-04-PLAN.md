### Task 1: Recursion Limit Handling

**What:** Implement recursion limit with proper error handling for GraphRecursionError.

**File to modify:** `src/agents/dev-agent.ts`
**File to create:** `src/agents/run-agent.ts`

**Implementation:**
```typescript
import { GraphRecursionError } from "@langchain/langgraph";
import { devAgent } from "./dev-agent";
import { devAgentConfig } from "../config";
import { logger } from "../logging";

export interface AgentResult {
  success: boolean;
  result?: unknown;
  error?: string;
  terminationReason?: "completed" | "recursion_limit" | "timeout" | "loop_limit" | "error";
}

export async function runAgentWithGuardrails(
  taskDescription: string,
  threadId: string
): Promise<AgentResult> {
  const startTime = Date.now();

  logger.log({
    action: "agent_start",
    context: {
      threadId,
      taskDescription,
      recursionLimit: devAgentConfig.recursionLimit,
      maxIterations: devAgentConfig.maxIterations,
    },
    outcome: "pending",
  });

  try {
    const result = await devAgent.invoke(
      {
        messages: [{ role: "user", content: taskDescription }],
      },
      {
        configurable: { thread_id: threadId },
        recursionLimit: devAgentConfig.recursionLimit, // NOT in configurable
      }
    );

    logger.log({
      action: "agent_complete",
      context: { threadId },
      outcome: "success",
      durationMs: Date.now() - startTime,
    });

    return {
      success: true,
      result,
      terminationReason: "completed",
    };
  } catch (error) {
    if (error instanceof GraphRecursionError) {
      logger.log({
        action: "agent_recursion_limit",
        context: { threadId, limit: devAgentConfig.recursionLimit },
        outcome: "failure",
        message: "Agent exceeded recursion limit",
        durationMs: Date.now() - startTime,
      });

      return {
        success: false,
        error: "Agent exceeded recursion limit",
        terminationReason: "recursion_limit",
      };
    }
    throw error;
  }
}
```

### Task 2: State Loop Counter Guard

**What:** Add loop counter checking in the agent's conditional routing.

**File to create:** `src/agents/guards.ts`

**Implementation:**
```typescript
import { END } from "@langchain/langgraph";
import { AIMessage } from "@langchain/core/messages";
import { logger } from "../logging";

export function createLoopGuard(maxIterations: number) {
  return function checkLoopLimit(state: { loopCount: number; messages: unknown[] }) {
    if (state.loopCount >= maxIterations) {
      logger.log({
        action: "loop_limit_reached",
        context: { loopCount: state.loopCount, maxIterations },
        outcome: "failure",
        message: `Loop limit ${maxIterations} reached`,
      });
      return END;
    }

    // Check if last message has tool calls
    const lastMessage = state.messages[state.messages.length - 1];
    if (lastMessage instanceof AIMessage && lastMessage.tool_calls?.length) {
      return "tools";
    }

    return END;
  };
}

export function incrementLoopCount(state: { loopCount: number }) {
  return { loopCount: state.loopCount + 1 };
}
```

**Note:** For createReactAgent, we wrap the agent invocation rather than modifying the graph. The loop counter provides observability and can trigger early termination via the runner.

### Task 3: Wall-Clock Timeout

**What:** Implement execution-level timeout that terminates agent if wall-clock time is exceeded.

**File to modify:** `src/agents/run-agent.ts`

**Implementation using AbortController:**
```typescript
export async function runAgentWithGuardrails(
  taskDescription: string,
  threadId: string,
  timeoutMs: number = devAgentConfig.timeoutMs
): Promise<AgentResult> {
  const startTime = Date.now();
  const abortController = new AbortController();

  // Set up wall-clock timeout
  const timeoutId = setTimeout(() => {
    abortController.abort();
  }, timeoutMs);

  logger.log({
    action: "agent_start",
    context: {
      threadId,
      taskDescription,
      timeoutMs,
    },
    outcome: "pending",
  });

  try {
    const result = await devAgent.invoke(
      {
        messages: [{ role: "user", content: taskDescription }],
      },
      {
        configurable: { thread_id: threadId },
        recursionLimit: devAgentConfig.recursionLimit,
        signal: abortController.signal,
      }
    );

    clearTimeout(timeoutId);

    logger.log({
      action: "agent_complete",
      context: { threadId },
      outcome: "success",
      durationMs: Date.now() - startTime,
    });

    return {
      success: true,
      result,
      terminationReason: "completed",
    };
  } catch (error) {
    clearTimeout(timeoutId);

    if (error instanceof Error && error.name === "AbortError") {
      logger.log({
        action: "agent_timeout",
        context: { threadId, timeoutMs },
        outcome: "failure",
        message: `Agent exceeded timeout of ${timeoutMs}ms`,
        durationMs: Date.now() - startTime,
      });

      return {
        success: false,
        error: `Agent exceeded timeout of ${timeoutMs}ms`,
        terminationReason: "timeout",
      };
    }

    if (error instanceof GraphRecursionError) {
      // ... existing recursion limit handling
    }

    logger.log({
      action: "agent_error",
      context: { threadId, error: String(error) },
      outcome: "failure",
      durationMs: Date.now() - startTime,
    });

    return {
      success: false,
      error: String(error),
      terminationReason: "error",
    };
  }
}
```
