export const CreateAgentActivityInputSchema = z.object({
  /** Agent session ID from the webhook payload */
  agentSessionId: z.string().min(1, "Agent session ID is required"),
  /** Activity type */
  type: z.enum(["thought", "action", "response", "error", "elicitation"]),
  /** Body text for thought, response, error, elicitation types */
  body: z.string().optional(),
  /** Action verb for action type (e.g., "Creating", "Reading") */
  action: z.string().optional(),
  /** Action parameter for action type (e.g., "branch feature/auth") */
  parameter: z.string().optional(),
  /** Action result for action type (optional completion message) */
  result: z.string().optional(),
  /** Ephemeral flag -- only valid for thought and action types */
  ephemeral: z.boolean().optional(),
});

export const CreateAgentActivityOutputSchema = z.object({
  success: z.boolean(),

**handleCreateAgentActivity(context, args, deps):**
1. Check permission for `create_agent_activity` via `checkLinearToolPermission`
2. Parse input with `CreateAgentActivityInputSchema.safeParse(args)`
3. Build the content payload based on `input.type`:
   - `thought`, `response`, `error`, `elicitation`: `{ type: input.type, body: input.body }` -- validate `body` is present, return error if missing
   - `action`: `{ type: "action", action: input.action, parameter: input.parameter, ...(input.result && { result: input.result }) }` -- validate `action` and `parameter` are present
4. Validate ephemeral constraint: if `input.ephemeral` is set and type is not `thought` or `action`, return error "ephemeral is only valid for thought and action types" (per Pitfall 4 from research)
5. Use `withTokenRefresh` from Plan 67-01 to wrap the SDK call:
   ```typescript
   const credentialStore = createLinearCredentialStore({ db: deps.db as any, logger: deps.logger });
   const result = await withTokenRefresh(
     async (client) => client.createAgentActivity({
       agentSessionId: input.agentSessionId,
       content: contentPayload,
       ...(input.ephemeral !== undefined && (input.type === "thought" || input.type === "action") && { ephemeral: input.ephemeral }),
     }),
     { credentialStore, workspaceId: deps.workspaceId, logger: deps.logger }
   );
   ```
6. Return `createToolResult(context, "Activity created", { success: result.success, type: input.type })`

**handleUpdateSessionState(context, args, deps):**
1. Check permission for `update_session_state`
2. Parse input with `UpdateSessionStateInputSchema.safeParse(args)`
3. Use `withTokenRefresh` to call `client.agentSessionUpdate(input.sessionId, { status: input.status })`
4. Return result

Note: The research says Linear tracks session state automatically from activity types. However, the executor needs explicit state updates for lifecycle boundaries (e.g., error state on conversation failure). Keep this tool for infrastructure use.
