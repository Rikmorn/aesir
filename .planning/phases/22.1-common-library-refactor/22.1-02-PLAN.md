Pattern from RESEARCH.md:
```typescript
// packages/integrations/linear/src/config.ts
import dotenvFlow from "dotenv-flow";
import { z } from "zod";

dotenvFlow.config({ silent: true, default_node_env: "development" });

const linearEnvSchema = z.object({
  PORT: z.coerce.number().default(3001),
  NODE_ENV: z.enum(["development", "test", "production"]).default("development"),
  LOG_LEVEL: z.enum(["debug", "info", "warn", "error"]).default("info"),
  DB_HOST: z.string().default("localhost"),
  DB_PORT: z.coerce.number().default(5432),
  DB_USER: z.string().default("temporal"),
  DB_PASSWORD: z.string().default("temporal"),
  DB_NAME: z.string().default("temporal"),
  CREDENTIAL_ENCRYPTION_KEY: z.string().length(64).optional(),
  LINEAR_CLIENT_ID: z.string().min(1, "LINEAR_CLIENT_ID required"),
  LINEAR_CLIENT_SECRET: z.string().min(1, "LINEAR_CLIENT_SECRET required"),
  LINEAR_WEBHOOK_SECRET: z.string().min(1, "LINEAR_WEBHOOK_SECRET required"),
  OAUTH_CALLBACK_URL: z.string().url().optional(),
});

const result = linearEnvSchema.safeParse(process.env);
if (!result.success) {
  console.error("\n[X] Linear environment validation failed:\n");
  const fieldErrors = result.error.flatten().fieldErrors;
  for (const [field, errors] of Object.entries(fieldErrors)) {
    if (errors && errors.length > 0) {
      console.error(`  - ${field}: ${errors.join(", ")}`);
    }
  }
  console.error("\nCheck your .env files or environment variables.\n");
  process.exit(1);
}

export const env = result.data;
export const config = { /* typed nested config */ } as const;
```
