```typescript
// packages/integrations/linear/src/config.ts
import dotenvFlow from "dotenv-flow";
import { z } from "zod";

dotenvFlow.config({ silent: true, default_node_env: "development" });

const linearEnvSchema = z.object({
  PORT: z.coerce.number().default(3001),
  NODE_ENV: z.enum(["development", "test", "production"]).default("development"),
  LOG_LEVEL: z.enum(["debug", "info", "warn", "error"]).default("info"),
  LINEAR_CLIENT_ID: z.string().min(1, "LINEAR_CLIENT_ID required"),
  // ... etc, using DATABASE_* names to match Docker
});

const result = linearEnvSchema.safeParse(process.env);
if (!result.success) {
  console.error("\n[X] Linear environment validation failed:\n");
  // ... format errors
  process.exit(1);
}

export const env = result.data;
export const config = { /* typed nested config */ } as const;
```
