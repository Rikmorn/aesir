```typescript
describe("Flow 1: Start -> Run -> Complete", () => {
  it("creates conversation, executes agent loop, transitions to completed, records events", async () => {
    mockAgentLoopCompletes(runAgentLoop as any);

    const convId = await executor.start({
      agentDefinitionId: "dev-agent",
      correlationKey: "FLOW1-001",
      initialMessage: "Implement auth module",
    });

    // Verify created as queued
    const initial = await executor.get(convId);
    expect(initial?.status).toBe("queued");

    // Start worker, wait for completion
    executor.startWorker();
    await waitForStatus(executor, convId, "completed", 10000);
    await executor.stopWorker();

    // Verify final state
    const completed = await executor.get(convId);
    expect(completed?.status).toBe("completed");

    // Verify events recorded
    const events = await eventLog.query(convId);
    const types = events.map(e => e.type);
    expect(types).toContain("agent.started");
    expect(types).toContain("agent.completed");

    // Verify session projection
    const session = await sessionProjection.getSession(convId);
    expect(session).not.toBeNull();
    expect(session?.status).toBe("completed");
  });
});
```

**Flow 5: Duplicate Start**

```typescript
describe("Flow 5: Duplicate Start", () => {
  it("returns same conversation ID for duplicate start (idempotent)", async () => {
    mockAgentLoopCompletes(runAgentLoop as any);

    const convId1 = await executor.start({
      agentDefinitionId: "dev-agent",
      correlationKey: "FLOW5-001",
      initialMessage: "First start",
    });

    const convId2 = await executor.start({
      agentDefinitionId: "dev-agent",
      correlationKey: "FLOW5-001",
      initialMessage: "Second start (should be idempotent)",
    });

    // Same ID returned (idempotent)
    expect(convId2).toBe(convId1);

    // Only one conversation exists
    const list = await executor.list({ agentDefinitionId: "dev-agent" });
    const matching = list.filter(c => c.id.startsWith("dev-agent-FLOW5-001"));
    expect(matching.length).toBe(1);
  });

  it("creates re-triggered conversation after terminal state", async () => {
    mockAgentLoopCompletes(runAgentLoop as any);

    // Start and complete first conversation
    const convId1 = await executor.start({
      agentDefinitionId: "dev-agent",
      correlationKey: "FLOW5-002",
      initialMessage: "First attempt",
    });

    executor.startWorker();
    await waitForStatus(executor, convId1, "completed", 10000);
    await executor.stopWorker();

    // Re-trigger after completion
    const convId2 = await executor.start({
      agentDefinitionId: "dev-agent",
      correlationKey: "FLOW5-002",
      initialMessage: "Second attempt",
    });

    // Different ID with -r2 suffix
    expect(convId2).not.toBe(convId1);
    expect(convId2).toMatch(/-r\d+$/);
  });
});
```

```typescript
describe("Flow 8: History Compaction", () => {
  it("compacts history when messages exceed threshold", async () => {
    // Create a special agent definition with very low pruneThreshold
    // Override the test agent registry or create a custom one for this test

    // The key insight: the worker loop calls historyManager.compact() before
    // passing messages to runAgentLoop. To test compaction, we need to:
    // 1. Create a conversation with many messages (pre-populate via DB)
    // 2. Let the worker pick it up
    // 3. Verify runAgentLoop received compacted messages

    let receivedMessages: unknown[] = [];
    const mockFn = runAgentLoop as ReturnType<typeof vi.fn>;
    mockFn.mockImplementation(async (options: any) => {
      receivedMessages = options.messages || [];
      return {
        status: "completed",
        output: "Done with compacted history",
        toolCallCount: 0,
        tokenCount: { input: 100, output: 50 },
        trace: [],
      };
    });

    // Start a conversation
    const convId = await executor.start({
      agentDefinitionId: "dev-agent",
      correlationKey: "FLOW8-001",
      initialMessage: "Start compaction test",
    });

    // Pre-populate with many messages via direct DB update
    // Generate messages that would exceed pruneThreshold token count
    // The dev-agent definition has pruneThreshold: 80000 tokens
    // For testing, we need enough messages. Alternatively, create a test agent
    // with a very low pruneThreshold (e.g., 100 tokens / 5 messages)
    const manyMessages = [
      { role: "user", content: "Start compaction test" },
    ];
    // Add messages with tool_use and tool_result pairs that will be pruned
    for (let i = 0; i < 30; i++) {
      manyMessages.push(
        { role: "assistant", content: [{ type: "tool_use", id: `tool_${i}`, name: "read_file", input: { path: `file${i}.ts` } }] },
        { role: "user", content: [{ type: "tool_result", tool_use_id: `tool_${i}`, content: "x".repeat(500) }] },
      );
    }

    // Update conversation messages directly
    // Note: this bypasses executor API but is necessary to set up preconditions
    await ctx.db.update(conversations)
      .set({ messages: manyMessages })
      .where(eq(conversations.id, convId));

    executor.startWorker();
    await waitForStatus(executor, convId, "completed", 15000);
    await executor.stopWorker();

    // Verify compaction happened: agent received fewer messages than original
    // Phase 1 pruning replaces tool results with descriptors
    // The exact count depends on protectedMessages config
    // Key assertion: at least some tool results were replaced with descriptors
    expect(receivedMessages.length).toBeLessThan(manyMessages.length);
  });
});
```
