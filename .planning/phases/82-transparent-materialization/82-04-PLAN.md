       // 4b. Attempt materialization if requested
       let materializationWarning: string | undefined;
       if (parsed.data.materialization?.type === "transparent" && deps.materializationAdapter) {
         // Resolve parent Linear issue ID from conversation context
         const parentIssueId = await resolveParentLinearIssueId(deps, ctx);

         const matResult = await deps.materializationAdapter.create({
           taskId: task.id,
           conversationId: ctx.correlationId,
           agentId: ctx.agentId,
           description,
           properties: parsed.data.materialization.properties ?? {},

         }

    4. Add helper function `resolveParentLinearIssueId`:
       ```typescript
       async function resolveParentLinearIssueId(
         deps: NonNullable<ToolContext["delegationDeps"]>,
         ctx: ToolContext,
       ): Promise<string | null> {
         try {
           // Check conversation's entityRef for a Linear issue
           const [conv] = await deps.db
             .select({ entity_ref: conversations.entity_ref })
             .from(conversations)
             .where(eq(conversations.id, ctx.correlationId))
             .limit(1);

           const entityRef = conv?.entity_ref as { entityType?: string; entityId?: string } | null;
           if (entityRef?.entityType === "linear_issue" && entityRef?.entityId) {
             return entityRef.entityId;
           }
           return null;
         } catch {
           return null; // Non-fatal
         }
       }
       ```
       Note: `conversations` table has `entity_ref` JSONB column. Import it if not already.
