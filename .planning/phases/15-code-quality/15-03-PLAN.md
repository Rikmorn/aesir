import { AppError, type ErrorMetadata, type RecoveryHint } from "@aesir/common";

// Database error codes
export type DatabaseErrorCode =
  | "PLT_DB_CONNECTION"
  | "PLT_DB_QUERY"
  | "PLT_DB_CONSTRAINT"
  | "PLT_DB_TIMEOUT";

export class DatabaseError extends AppError {
  readonly code: DatabaseErrorCode;

  constructor(
    code: DatabaseErrorCode,
    message: string,
    options?: { cause?: Error; metadata?: ErrorMetadata; recovery?: RecoveryHint }
  ) {
    super(message, options);
    this.code = code;
  }

  override get httpStatus(): number {
    switch (this.code) {
      case "PLT_DB_TIMEOUT": return 504;
      default: return 500;
    }
  }
}

Update the interface methods to return ResultAsync:
```typescript
import { ResultAsync, ok, err, fromPromise } from "neverthrow";
import { CleanupError } from "../errors/index.js";

export interface CleanupResult {
  checkpointsDeleted: number;
  webhooksDeleted: number;
  executionsDeleted: number;
}

export interface CleanupService {
  cleanupOldData(retentionDays: number): ResultAsync<CleanupResult, CleanupError>;
  health(): Promise<{ healthy: boolean; latencyMs: number }>;
  close(): Promise<void>;
}
```

Update implementation to wrap with fromPromise and return CredentialError on failure:
```typescript
return {
  cleanupOldData(retentionDays): ResultAsync<CleanupResult, CleanupError> {
    return fromPromise(
      cleanupImpl(db, logger, retentionDays),
      (error) => {
        logger.error({ err: error, retentionDays }, "Cleanup operation failed");
        return new CleanupError(
          "PLT_CLEANUP_CHECKPOINTS",
          "Failed to clean up old data",
          {
            cause: error instanceof Error ? error : new Error(String(error)),
            metadata: { retentionDays },
          }
        );
      }
    );
  },
  // health() and close() stay Promise-based
};
```
