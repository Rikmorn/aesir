import { AppError, type ErrorMetadata, type RecoveryHint } from "@aesir/common";

// Linear error codes
export type LinearErrorCode =
  | "INT_LINEAR_NOT_FOUND"
  | "INT_LINEAR_RATE_LIMIT"
  | "INT_LINEAR_AUTH"
  | "INT_LINEAR_API"
  | "INT_LINEAR_VALIDATION";

export class LinearError extends AppError {
  readonly code: LinearErrorCode;

  constructor(
    code: LinearErrorCode,
    message: string,
    options?: { cause?: Error; metadata?: ErrorMetadata; recovery?: RecoveryHint }
  ) {
    super(message, options);
    this.code = code;
  }

  override get httpStatus(): number {
    switch (this.code) {
      case "INT_LINEAR_NOT_FOUND": return 404;
      case "INT_LINEAR_RATE_LIMIT": return 429;
      case "INT_LINEAR_AUTH": return 401;
      default: return 500;
    }
  }
}

2. Update factory to wrap operations with fromPromise:
```typescript
export function createCredentialStore(options: CredentialStoreOptions): CredentialStore {
  // Keep factory validation - throws are OK here (startup fail-fast)
  if (!options.db) throw new Error("db is required for CredentialStore");
  if (!options.logger) throw new Error("logger is required for CredentialStore");

  const { db, logger, encryptionKey } = options;

  return {
    store(input): ResultAsync<string, CredentialError> {
      return fromPromise(
        storeCredentialImpl(db, logger, encryptionKey, input),
        (error) => {
          logger.error({ err: error, ...input }, "Failed to store credential");
          return new CredentialError(
            "INT_CRED_STORE",
            "Failed to store credential",
            {
              cause: error instanceof Error ? error : new Error(String(error)),
              metadata: { workspaceId: input.workspaceId, provider: input.provider },
            }
          );
        }
      );
    },
    // Similar pattern for get, getByProvider, delete
    // health() and close() remain Promise-based (not service boundaries)
  };
}
```
