```typescript
import { z } from "zod";

// AgentSession object within the webhook payload
export const AgentSessionSchema = z.object({
  id: z.string(),
  issueId: z.string(),
  status: z.enum(["pending", "active", "completed", "error"]).optional(),
  url: z.string().url().optional(),
});

// Full webhook payload from Linear
export const AgentSessionPayloadSchema = z.object({
  type: z.literal("AgentSessionEvent"),
  action: z.enum(["created", "prompted"]),
  webhookTimestamp: z.number(),
  webhookId: z.string(),
  agentSession: AgentSessionSchema,
});

export type AgentSessionPayload = z.infer<typeof AgentSessionPayloadSchema>;
export type AgentSession = z.infer<typeof AgentSessionSchema>;

// Helper to validate and parse
export function parseAgentSessionPayload(
  rawBody: string
): { success: true; data: AgentSessionPayload } | { success: false; error: z.ZodError } {
  try {
    const json = JSON.parse(rawBody);
    return AgentSessionPayloadSchema.safeParse(json);
  } catch (e) {
    // JSON parse error - create a synthetic Zod error
    return {
      success: false,
      error: new z.ZodError([
        {
          code: z.ZodIssueCode.custom,
          message: "Invalid JSON",
          path: [],
        },
      ]),
    };
  }
}
```

import { z } from "zod";

// Pull request object
export const PullRequestSchema = z.object({
  number: z.number(),
  title: z.string(),
  body: z.string().nullable(),
  head: z.object({
    ref: z.string(),
    sha: z.string(),
  }),
  base: z.object({
    ref: z.string(),
  }),
  html_url: z.string().url(),
});

// Review object
export const ReviewSchema = z.object({
  id: z.number(),
  user: z.object({
    login: z.string(),
  }),
  body: z.string().nullable(),
  state: z.enum(["approved", "changes_requested", "commented", "dismissed", "pending"]),
  submitted_at: z.string().datetime().nullable(),
});

// Repository object
export const RepositorySchema = z.object({
  full_name: z.string(),
  owner: z.object({
    login: z.string(),
  }),
  name: z.string(),
});

// Full PR review webhook payload
export const PRReviewPayloadSchema = z.object({
  action: z.enum(["submitted", "edited", "dismissed"]),
  review: ReviewSchema,
  pull_request: PullRequestSchema,
  repository: RepositorySchema,
});

export type PRReviewPayload = z.infer<typeof PRReviewPayloadSchema>;
export type PullRequest = z.infer<typeof PullRequestSchema>;
export type Review = z.infer<typeof ReviewSchema>;
export type Repository = z.infer<typeof RepositorySchema>;

export function parsePRReviewPayload(
  rawBody: string
): { success: true; data: PRReviewPayload } | { success: false; error: z.ZodError } {
  try {

    return PRReviewPayloadSchema.safeParse(json);
  } catch (e) {
    return {
      success: false,
      error: new z.ZodError([
        {
          code: z.ZodIssueCode.custom,
          message: "Invalid JSON",
          path: [],
        },
      ]),
    };
  }
}
