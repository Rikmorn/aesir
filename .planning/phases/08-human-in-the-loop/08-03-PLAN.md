import * as wf from '@temporalio/workflow';
import { proxyActivities } from '@temporalio/workflow';
import type { ApprovalDecision, WorkflowConfig, ApprovalStatus } from '../types.js';
import { approvalSignal, changesRequestedSignal } from '../signals.js';

// Import activity types for proxyActivities
import type * as activities from '../activities/index.js';

// Configure activities with appropriate timeouts
const { executeDevWorkflow, mergePRActivity, sendApprovalRequestActivity, sendStatusUpdateActivity, updateLinearStatusActivity } = proxyActivities<typeof activities>({
  startToCloseTimeout: '30 minutes',  // Dev workflow can take a while
  retry: {
    maximumAttempts: 3,
    initialInterval: '1 second',
    backoffCoefficient: 2,
  },
});

// Query for checking approval status
export const approvalStatusQuery = wf.defineQuery<ApprovalStatus>('approvalStatus');

/**
 * Workflow input configuration
 */
export interface ApprovalWorkflowInput {
  taskId: string;
  config: WorkflowConfig;
  slackChannel: string;
  // Injected clients will be passed to activities
  // (Temporal workflows can't have non-serializable inputs,
  // so clients are created inside activities or passed differently)
}

/**
 * Workflow result
 */
export interface ApprovalWorkflowResult {
  success: boolean;
  prNumber?: number;
  outcome: 'approved' | 'rejected' | 'timeout' | 'failed';
  message?: string;
}

/**
 * PR Approval Workflow
 *
 * Orchestrates the complete approval flow:
 * 1. Run Dev Agent to create PR
 * 2. Send Slack notification for review
 * 3. Wait for approval signal (with timeout)
 * 4. On approval: merge PR and update Linear
 * 5. On rejection: update Linear status
 * 6. On changes requested: re-run dev workflow with feedback
 *
 * The workflow survives process restarts and can wait indefinitely
 * for human approval.
 */
export async function prApprovalWorkflow(
  input: ApprovalWorkflowInput
): Promise<ApprovalWorkflowResult> {
  const { taskId, config, slackChannel } = input;

  // Workflow state
  let decision: ApprovalDecision | null = null;
  let changesRequested: { reviewer: string; feedback: string } | null = null;
  let prNumber: number | undefined;
  let currentStatus: ApprovalStatus['status'] = 'running';

  // Set up signal handlers
  wf.setHandler(approvalSignal, (input: ApprovalDecision) => {
    wf.log.info('Received approval decision', { approved: input.approved, reviewer: input.reviewer });
    decision = input;
  });

  wf.setHandler(changesRequestedSignal, (input: { reviewer: string; feedback: string }) => {
    wf.log.info('Received changes requested', { reviewer: input.reviewer });
    changesRequested = input;
  });

  // Query handler for status checks
  wf.setHandler(approvalStatusQuery, (): ApprovalStatus => ({
    taskId,
    prNumber,
    status: currentStatus,
    decision,
  }));

  // Maximum iterations for changes-requested loop
  const maxFeedbackIterations = config.maxFeedbackIterations ?? 3;
  let feedbackIteration = 0;

  while (feedbackIteration < maxFeedbackIterations) {
    feedbackIteration++;

    // Step 1: Run Dev Agent workflow
    currentStatus = 'running';
    wf.log.info(`Running dev workflow (iteration ${feedbackIteration})`);

    try {
      // Note: In real implementation, dependencies need to be passed differently
      // since Temporal requires serializable inputs. This will need adaptation.
      const devResult = await executeDevWorkflow(taskId, {} as any); // TODO: Fix dependency injection

      if (!devResult.success || !devResult.prNumber) {
        return {
          success: false,
          outcome: 'failed',
          message: devResult.error ?? 'Dev workflow failed to create PR',
        };
      }

      prNumber = devResult.prNumber;
    } catch (error) {
      return {
        success: false,
        outcome: 'failed',
        message: error instanceof Error ? error.message : 'Dev workflow failed',
      };
    }

    // Step 2: Send Slack notification
    currentStatus = 'awaiting_approval';
    wf.log.info('Sending approval request notification');

    await sendApprovalRequestActivity(
      {} as any, // TODO: Client injection
      {
        type: 'approval_needed',
        taskId,
        title: `PR #${prNumber} ready for review`,
        summary: `Dev Agent completed task ${taskId}`,
        prUrl: config.prUrl ?? `https://github.com/${config.owner}/${config.repo}/pull/${prNumber}`,
      },
      slackChannel
    );

    // Step 3: Wait for signal OR timeout
    const approvalTimeoutDays = config.approvalTimeoutDays ?? 7;
    wf.log.info(`Waiting for approval (timeout: ${approvalTimeoutDays} days)`);

    // Reset for this iteration
    decision = null;
    changesRequested = null;

    const receivedSignal = await wf.condition(
      () => decision !== null || changesRequested !== null,
      `${approvalTimeoutDays} days`
    );

    // Check what happened
    if (!receivedSignal) {
      // Timeout - no signal received
      wf.log.warn('Approval timed out');
      currentStatus = 'timeout';
      return {
        success: false,
        prNumber,
        outcome: 'timeout',
        message: `Approval timed out after ${approvalTimeoutDays} days`,
      };
    }

    if (changesRequested) {
      // Changes requested - loop back
      wf.log.info(`Changes requested by ${changesRequested.reviewer}, iteration ${feedbackIteration}`);
      currentStatus = 'running';
      // Continue loop to re-run dev workflow with feedback
      // The agent will pick up PR comments as feedback
      continue;
    }

    if (decision) {
      if (decision.approved) {
        // Step 4a: Approved - merge PR and update Linear
        wf.log.info(`PR approved by ${decision.reviewer}`);
        currentStatus = 'approved';

        await mergePRActivity(
          {} as any, // TODO: Client injection
          {
            owner: config.owner,
            repo: config.repo,
            pullNumber: prNumber,
            mergeMethod: 'squash',
          }
        );

        // Update Linear to configurable completion status
        await updateLinearStatusActivity(
          {} as any, // TODO: Client injection
          taskId,
          config.completionStatus
        );

        // Ensure all signal handlers complete before returning
        await wf.condition(wf.allHandlersFinished);

        return {
          success: true,
          prNumber,
          outcome: 'approved',
          message: `PR merged by ${decision.reviewer}`,
        };
      } else {
        // Step 4b: Rejected
        wf.log.info(`PR rejected by ${decision.reviewer}: ${decision.comment}`);
        currentStatus = 'rejected';

        // Update Linear back to Todo or custom rejected status
        await updateLinearStatusActivity(
          {} as any, // TODO: Client injection
          taskId,
          'Todo' // Or could be configurable
        );

        await wf.condition(wf.allHandlersFinished);

        return {
          success: false,
          prNumber,
          outcome: 'rejected',
          message: decision.comment ?? `Rejected by ${decision.reviewer}`,
        };
      }
    }
  }

  // Exceeded max feedback iterations
  wf.log.warn(`Max feedback iterations (${maxFeedbackIterations}) exceeded`);
  await wf.condition(wf.allHandlersFinished);

  return {
    success: false,
    prNumber,
    outcome: 'failed',
    message: `Max feedback iterations (${maxFeedbackIterations}) exceeded`,
  };
}
