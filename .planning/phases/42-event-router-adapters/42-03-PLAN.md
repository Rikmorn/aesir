/**
 * Route result from EventRouter.handle().
 * Discriminated union on the `action` field.
 */
export type EventRouterRouteResult =
  | { action: "start"; agentDefinitionId: string; conversationId: string; correlationKey: string; message: string; event: IncomingEvent }
  | { action: "signal"; conversationId: string; signal: Signal; event: IncomingEvent }
  | { action: "slow_path"; event: IncomingEvent }
  | { action: "ignore"; reason: string };

/**
 * Options for creating an EventRouter.
 */
export interface EventRouterOptions {
  /** Agent registry for loading trigger rules */
  agentRegistry: AgentRegistry;
  /** Logger instance */
  logger: PinoLogger;
}

/**
 * EventRouter - Matches IncomingEvents against agent trigger rules.
 *
 * Produces routing decisions (start, signal, ignore, slow_path) without
 * executing them. The caller is responsible for executing the decision
 * via ConversationExecutor, enrichment, or slow-path LLM routing.
 */
export interface EventRouter {
  /**
   * Initialize the router by loading start rules from AgentRegistry.
   * Must be called before handle(). Can be called again to refresh rules.
   */
  loadStartRules(): Promise<void>;

  /**
   * Route an IncomingEvent to a routing decision.
   */
  handle(event: IncomingEvent): EventRouterRouteResult;
}
```

import type { IncomingEvent } from "../adapters/types.js";

  AgentDefinition,
  AgentRegistry,
  EventRouter,
  EventRouterOptions,
  EventRouterRouteResult,
  Signal,
} from "./types.js";
```

Implementation details:

1. **Start rules** -- loaded from `agentRegistry.list()` at init. Each agent definition's `triggers` array maps `trigger.event` to `agentDefinitionId`. Stored as `Map<string, string>` (eventType -> agentDefinitionId).

2. **Signal routing map** -- Maps domain-language signal types to agent definition IDs for conversation ID construction:
   ```typescript
   const SIGNAL_AGENT_MAP: Record<string, string> = {
     "approval": "dev-agent",
     "escalation_resolved": "dev-agent",
     "pr_merged": "dev-agent",
     "pr_closed": "dev-agent",
     "pr_review": "dev-agent",
     "user_reply": "product-agent",
     "cancel": "dev-agent", // admin cancellation
   };
   ```

3. **handle(event)** routing logic (evaluated in order):

   a. **Ignore check**: If `IGNORE_EVENT_TYPES.has(event.type)`, return `{ action: "ignore", reason: "Event type explicitly ignored: {event.type}" }`.

   b. **Start rule check**: If `startRules.has(event.type)` AND `event.correlationKey` exists:
      - `agentDefinitionId = startRules.get(event.type)`
      - `conversationId = {agentDefinitionId}-{event.correlationKey}`
      - Return `{ action: "start", agentDefinitionId, conversationId, correlationKey: event.correlationKey, message: event.message ?? JSON.stringify(event.data), event }`

   c. **Signal rule check**: If `SIGNAL_AGENT_MAP[event.type]` exists AND `event.correlationKey` exists:
      - `agentId = SIGNAL_AGENT_MAP[event.type]`
      - `conversationId = {agentId}-{event.correlationKey}`
      - Build `Signal` object: `{ type: event.type, data: event.data, message: event.message, source: event.source, deduplicationId: event.deduplicationId }`
      - Return `{ action: "signal", conversationId, signal, event }`

   d. **Fallthrough**: Return `{ action: "slow_path", event }`.

4. **loadStartRules()**: Calls `agentRegistry.list()`, iterates definitions, builds startRules Map.

5. The `handle()` method is **synchronous** -- it is a pure routing decision with no I/O. Only `loadStartRules()` is async (reads from registry).
