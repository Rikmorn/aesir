**1. Replace page.tsx** -- Replace the current placeholder with a full async server component page.

Follow the exact pattern from `tools/page.tsx`: async server component, `Promise.all` for parallel data loading, pass typed data as props to presentational components.

```typescript
import { StatCards } from "@/components/overview/stat-cards";
import { ActiveConversations } from "@/components/overview/active-conversations";
import { WorkerStatus } from "@/components/overview/worker-status";
import { RecentErrors } from "@/components/overview/recent-errors";
import { TokenUsage } from "@/components/overview/token-usage";
import { fetchWorkerStatus } from "@/lib/agent-service";
import {
  getConversationStatusCounts,
  getActiveConversations,
  getRecentErrors,
  getTokenUsageByAgent,
} from "@/services/overview";

export default async function OverviewPage() {
  const [statusCounts, activeConversations, workerStatus, recentErrors, tokenUsage] =
    await Promise.all([
      getConversationStatusCounts(),
      getActiveConversations(),
      fetchWorkerStatus(),
      getRecentErrors(),
      getTokenUsageByAgent(),
    ]);

  return (
    <main className="container mx-auto px-4 py-8">
      {/* Header */}
      <div className="mb-6">
        <h1 className="text-2xl font-bold tracking-tight">System Overview</h1>
        <p className="mt-1 text-sm text-muted-foreground">
          Real-time pulse check on agent activity, health, and resource usage
        </p>
      </div>

      {/* Content */}
      <div className="space-y-6">
        {/* Status summary cards */}
        <StatCards counts={statusCounts} />

        {/* Active conversations + Worker status */}
        <div className="grid gap-4 lg:grid-cols-2">
          <ActiveConversations conversations={activeConversations} />
          <WorkerStatus status={workerStatus} />
        </div>

        {/* Recent errors + Token usage */}
        <div className="grid gap-4 lg:grid-cols-2">
          <RecentErrors errors={recentErrors} />
          <TokenUsage data={tokenUsage} />
        </div>
      </div>
    </main>
  );
}
```

Layout follows CONTEXT.md decisions exactly:
- `space-y-6` between sections
- `gap-4` within grids
- `lg:grid-cols-2` for two-column layout
- `container mx-auto px-4 py-8` for full-width container
- `text-2xl font-bold` title + `text-muted-foreground` subtitle
