    router.get("/states", async (req, res) => {
      try {
        const agents = await agentRegistry.list();
        const allStates = await scheduleRegistry.getAllScheduleStates();

        // Build response with nextRunAt computed from cron expression
        const result: ScheduleStateResponse[] = [];
        for (const agent of agents) {
          if (!agent.schedules) continue;
          for (const schedule of agent.schedules) {
            const state = allStates.find(
              s => s.agentId === agent.id && s.scheduleName === schedule.name

            // Compute next run time using cron-parser
            let nextRunAt: string | null = null;

              const { CronExpressionParser } = await import("cron-parser");

                tz: schedule.timezone ?? "UTC",
              });
              nextRunAt = interval.next().toISOString();
            } catch { /* invalid cron -- skip */ }

            // Derive health: healthy, failed, or missed
            let health: "healthy" | "failed" | "missed" = "healthy";
            if (state?.lastRunOutcome === "failed" || state?.lastRunOutcome === "cancelled") {
              health = "failed";
            }
            // Missed: nextRunAt is in the past and no recent conversation
            // Simplified: if nextRunAt is past and last run was more than 2 cron intervals ago
            // For now, just use failed/healthy. Missed detection is complex and can be enhanced later.

            result.push({

    Per context decision: "Overview page: 'Upcoming Schedules' card showing next 3-5 scheduled runs across all agents, sorted by time."

    Server component that receives data as props (pre-fetched in page.tsx). Or client component if it needs auto-refresh. Use the same pattern as other overview cards.

    ```tsx
    import Link from "next/link";
    import { Clock, Play } from "lucide-react";
    import type { ScheduleState } from "@/lib/agent-service";

    interface UpcomingSchedulesCardProps {
      scheduleStates: ScheduleState[];
    }

    export function UpcomingSchedulesCard({ scheduleStates }: UpcomingSchedulesCardProps) {
      // Sort by nextRunAt ascending, take first 5
      const upcoming = scheduleStates
        .filter(s => s.nextRunAt)
        .sort((a, b) => new Date(a.nextRunAt!).getTime() - new Date(b.nextRunAt!).getTime())
        .slice(0, 5);

      if (upcoming.length === 0) {
        return (
          <div className="rounded-lg border bg-card p-4">
            <h3 className="mb-2 text-sm font-medium text-foreground flex items-center gap-1.5">
              <Clock className="h-4 w-4 text-muted-foreground" />
              Upcoming Schedules
            </h3>
            <p className="text-xs text-muted-foreground">No scheduled agents configured</p>
          </div>
        );
      }

      return (
        <div className="rounded-lg border bg-card p-4">
          <h3 className="mb-3 text-sm font-medium text-foreground flex items-center gap-1.5">
            <Clock className="h-4 w-4 text-muted-foreground" />
            Upcoming Schedules
          </h3>
          <div className="space-y-2">
            {upcoming.map((schedule) => (
              <div key={`${schedule.agentId}-${schedule.scheduleName}`} className="flex items-center justify-between text-xs">
                <div className="flex items-center gap-2 min-w-0">
                  <Link
                    href={`/agents/${schedule.agentId}`}
                    className="font-medium text-foreground hover:underline truncate"
                  >
                    {schedule.agentName ?? schedule.agentId}
                  </Link>
                  <span className="text-muted-foreground truncate">{schedule.scheduleName}</span>
                </div>
                <div className="flex items-center gap-2 shrink-0">
                  {/* Health indicator dot */}
                  <span className={`h-1.5 w-1.5 rounded-full ${
                    schedule.health === "healthy" ? "bg-emerald-500" :
                    schedule.health === "failed" ? "bg-red-500" : "bg-amber-500"
                  }`} />
                  <span className="text-muted-foreground tabular-nums">
                    {formatRelativeTime(schedule.nextRunAt!)}
                  </span>
                </div>
              </div>
            ))}
          </div>
        </div>
      );
    }

    function formatRelativeTime(isoDate: string): string {
      const now = new Date();
      const target = new Date(isoDate);
      const diffMs = target.getTime() - now.getTime();
      if (diffMs < 0) return "overdue";
      const hours = Math.floor(diffMs / (1000 * 60 * 60));
      const minutes = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60));
      if (hours > 24) return `in ${Math.floor(hours / 24)}d`;
      if (hours > 0) return `in ${hours}h ${minutes}m`;
      return `in ${minutes}m`;
    }
    ```
