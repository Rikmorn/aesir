    **1. Create `schedule-trigger.ts`:**

    Per locked decision: `POST /api/schedules/:agentId/:scheduleName/trigger` with `{ force?: boolean }` body.

    ```typescript
    import type { PinoLogger } from "@aesir/platform";
    import { Router } from "express";
    import type { AgentRegistry, ScheduleRegistry } from "../../framework/types.js";

    export interface ScheduleTriggerRouterOptions {
      agentRegistry: AgentRegistry;
      scheduleRegistry: ScheduleRegistry;
      logger: PinoLogger;
    }

    export function createScheduleTriggerRouter(options: ScheduleTriggerRouterOptions): Router {
      const { agentRegistry, scheduleRegistry, logger } = options;
      const router = Router();

      router.post("/:agentId/:scheduleName/trigger", async (req, res) => {
        const { agentId, scheduleName } = req.params;
        const { force } = req.body ?? {};

        try {
          // Validate agent exists and has this schedule
          const agent = await agentRegistry.get(agentId);
          if (!agent) {
            return res.status(404).json({ error: `Agent not found: ${agentId}` });
          }

          const schedule = agent.schedules?.find(s => s.name === scheduleName);
          if (!schedule) {
            return res.status(404).json({ error: `Schedule not found: ${scheduleName} on agent ${agentId}` });
          }

          // Build correlationKey -- force uses timestamp suffix to bypass idempotency
          const correlationKey = force
            ? `${agentId}:${scheduleName}:${Date.now()}`
            : `${agentId}:${scheduleName}`;

          // Build schedule context
          const context = await scheduleRegistry.buildScheduleContext(agentId, scheduleName, "manual");

          // Create synthetic event
          const syntheticEvent = {
            type: "schedule.triggered" as const,
            source: "scheduler" as const,
            correlationKey,
            data: {
              agentId,
              scheduleName,
              cron: schedule.cron,
              trigger: "manual" as const,
            },
            message: context + "\n\nManual trigger: " + scheduleName,
          };

          // Route through the event pipeline
          // The schedule registry's event handler is already set up
          // But the manual trigger needs its own path to routeEvent
          // Accept routeEventFn in options or call scheduleRegistry's handler
          // Actually: we need the routeEvent function here too.
          // Add routeEventFn to options.

          // NOTE TO EXECUTOR: The manual trigger needs to call the same routeEvent pipeline.
          // Add a `routeEventFn` callback to ScheduleTriggerRouterOptions that mirrors
          // what main.ts uses. In main.ts, pass: `routeEventFn: (event) => routeEvent(event, routeEventDeps)`

          await options.routeEventFn(syntheticEvent);

          logger.info(
            { agentId, scheduleName, force: !!force, correlationKey },
            "Manual schedule trigger fired",
          );

          res.json({
            triggered: true,
            correlationKey,
            force: !!force,
          });
        } catch (error) {
          logger.error(
            { err: error, agentId, scheduleName },
            "Manual schedule trigger failed",
          );
          res.status(500).json({ error: "Internal server error" });
        }
      });

      return router;
    }
    ```

    Add `routeEventFn: (event: IncomingEvent) => Promise<void>` to `ScheduleTriggerRouterOptions`. Import `IncomingEvent` from the adapters types.

    **Overlap handling for non-force triggers:**
    The existing `executor.start()` idempotency handles this -- if a conversation with the same correlationKey is active, it returns the existing ID. But per context decision, we want to actively detect and report the skip. Two approaches:
    1. Pre-check: query conversation status before routing (adds complexity to the API)
    2. Let start() handle it and accept the idempotent behavior (simpler)

    For the API, approach 2 is fine for non-force triggers. The conversation won't be duplicated. If the user wants to know if it was skipped, the response can check if the returned conversation was already active. But for simplicity, just route the event and return success -- the EventRouter/executor handles the rest. If a skip is needed at the API level (e.g., to return 409), add the pre-check. Per context: "Default respects skip policy -- if previous run active, skip with toast notification." This suggests the API should return 409 (skip) rather than 200. So add the pre-check:
