   ```typescript
   export async function productAgentConversationWorkflow(
     input: ProductAgentWorkflowInput
   ): Promise<ProductAgentWorkflowResult> {
     const { threadTs, channelId, initialMessage, userId } = input;

     // Workflow state
     const state = {
       userReply: null as string | null,
       cancelRequested: false,
       phase: 'pending' as ProductAgentPhase,
     };

     // Set up signal handlers
     setHandler(userReplySignal, (reply: string) => {
       log.info('Received user reply', { threadTs, userId });
       state.userReply = reply;
     });

     setHandler(cancelConversationSignal, () => {
       log.info('Received cancellation', { threadTs, userId });
       state.cancelRequested = true;
     });

     // Main conversation loop
     let iteration = 0;
     const maxIterations = 20; // Prevent infinite loops

     while (iteration < maxIterations && !state.cancelRequested) {
       iteration++;
       state.phase = 'running';

       // Run agent reasoning via activity
       const agentResult = await runProductAgentActivity({
         threadTs,
         message: state.userReply ?? initialMessage,
       });

       // Send response to Slack
       await sendSlackReplyActivity(channelId, threadTs, agentResult.response);

       // Terminal states
       if (agentResult.phase === 'complete') {
         state.phase = 'complete';
         await condition(allHandlersFinished);
         return { success: true, phase: 'complete', issueId: agentResult.issueId };
       }

       if (agentResult.phase === 'declined') {
         state.phase = 'declined';
         await condition(allHandlersFinished);
         return { success: true, phase: 'declined' };
       }

       // Waiting for user input
       if (agentResult.phase === 'clarifying' || agentResult.phase === 'confirming') {
         state.phase = 'awaiting_reply';
         state.userReply = null;

         // Wait 24h for reply
         const received = await condition(
           () => state.userReply !== null || state.cancelRequested,
           '24 hours'
         );

         if (!received && !state.cancelRequested) {
           // Send reminder
           await sendSlackReplyActivity(
             channelId, threadTs,
             'Still want to create this feature? Reply here to continue, or say "nevermind" to cancel.'
           );

           // Wait 48h more (72h total)
           const receivedAfterReminder = await condition(
             () => state.userReply !== null || state.cancelRequested,
             '48 hours'
           );

           if (!receivedAfterReminder) {
             await sendSlackReplyActivity(
               channelId, threadTs,
               'This conversation has timed out. Feel free to @mention me again if you need help!'
             );
             state.phase = 'timeout';
             await condition(allHandlersFinished);
             return { success: false, phase: 'timeout' };
           }
         }
       }

       // Check cancellation
       if (state.cancelRequested) {
         state.phase = 'cancelled';
         await sendSlackReplyActivity(
           channelId, threadTs,
           'No problem! Let me know if you change your mind.'
         );
         await condition(allHandlersFinished);
         return { success: false, phase: 'cancelled' };
       }
     }

     // Max iterations reached
     await condition(allHandlersFinished);
     return { success: false, phase: 'timeout' };
   }
   ```
