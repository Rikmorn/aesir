   ```typescript
   import { PostgresSaver } from '@langchain/langgraph-checkpoint-postgres';
   import { getDatabase } from '@aesir/platform';
   import { createPinoLogger } from '@aesir/common';

   let checkpointerInstance: PostgresSaver | null = null;

   /**
    * Create or return singleton PostgreSQL checkpointer for product-agent.
    * Uses agents.checkpoints table for state persistence.
    */
   export async function createProductAgentCheckpointer(): Promise<PostgresSaver> {
     if (checkpointerInstance) {
       return checkpointerInstance;
     }

     const logger = createPinoLogger({ component: 'product-agent:checkpointer' });
     const db = getDatabase();

     logger.info('Initializing PostgreSQL checkpointer');

     // PostgresSaver will create tables if they don't exist
     // Table name: agents.checkpoints (uses schema prefix)
     const checkpointer = PostgresSaver.fromConnString(
       process.env.DATABASE_URL!,
       {
         // Use agents schema for isolation
         schema: 'agents',
       }
     );

     // Setup creates the checkpoint tables
     await checkpointer.setup();

     checkpointerInstance = checkpointer;
     logger.info('PostgreSQL checkpointer ready');

     return checkpointer;
   }

   /**
    * Get existing checkpointer instance (must be created first).
    * Throws if not initialized.
    */
   export function getProductAgentCheckpointer(): PostgresSaver {
     if (!checkpointerInstance) {
       throw new Error('Checkpointer not initialized. Call createProductAgentCheckpointer first.');
     }
     return checkpointerInstance;
   }
   ```

   import { getProductAgentCheckpointer } from '../../product-agent/checkpointer.js';

   export async function runProductAgentActivity(
     input: RunProductAgentActivityInput
   ): Promise<RunProductAgentActivityOutput> {
     const { threadTs, message } = input;

     const logger = createPinoLogger({ component: 'agents:temporal:product-agent-activity' });

     logger.info({ threadTs }, 'Running product agent');

     // Get checkpointer (must be initialized at worker startup)
     const checkpointer = getProductAgentCheckpointer();

     // Create graph with checkpointer for conversation history
     const graph = createProductAgentGraph({
       teamId: getConfig().linear.teamId,
       checkpointer, // Enables conversation history persistence
     });

     // Convert message to LangGraph input
     const humanMessage = new HumanMessage(message);

     // Invoke graph with thread_ts as thread_id
     // Checkpointer automatically loads/saves state for this thread
     const result = await graph.invoke(
       { messages: [humanMessage] },
       {
         configurable: {
           thread_id: threadTs,
         },
       }
     );

     // Extract last AI message as response
     const lastMessage = result.messages[result.messages.length - 1];
     const response = extractMessageContent(lastMessage);

     // Get created task info if any
     const createdTask = result.createdTasks[0];

     logger.info(
       { threadTs, phase: result.phase, hasIssue: !!createdTask },
       'Product agent completed'
     );

     return {
       response,
       phase: result.phase,
       issueId: createdTask?.id,
       issueIdentifier: createdTask?.identifier,
     };
   }

   function extractMessageContent(message: BaseMessage): string {
     if (typeof message.content === 'string') return message.content;
     if (Array.isArray(message.content)) {
       return message.content
         .map(c => typeof c === 'string' ? c : 'text' in c ? c.text : '')
         .join('');
     }
     return '';
   }
   ```

   /**
    * Send a reply to a Slack thread as a Temporal activity.
    *
    * @param channelId - Slack channel ID
    * @param threadTs - Thread timestamp to reply to
    * @param text - Message text
    * @returns Message result with success status
    */
   export async function sendSlackReplyActivity(
     channelId: string,
     threadTs: string,
     text: string,
   ): Promise<MessageResult> {
     logger.info(
       { channelId, threadTs },
       'Sending Slack thread reply'
     );

     // Call Slack integration service via MCP
     const result = await callMcpTool<MessageResult>({
       integration: 'slack',
       tool: 'reply_to_thread',
       params: {
         channel: channelId,
         thread_ts: threadTs,
         text,
       },
       agentId: 'product-agent',
       correlationId: `slack-reply-${threadTs}-${Date.now()}`,
     });

     return result;
   }
   ```
