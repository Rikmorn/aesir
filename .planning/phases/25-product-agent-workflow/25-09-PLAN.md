/**
 * Product Agent Integration Tests
 *
 * Tests the complete flow from Slack event to Linear issue creation.
 * Requires running infrastructure (Docker Compose).
 *
 * Run with: pnpm --filter @aesir/agents test:integration
 */

import { describe, it, expect, beforeAll, afterAll } from 'vitest';

// These tests require Docker Compose infrastructure
// Skip if not in integration test mode
const isIntegrationTest = process.env.TEST_MODE === 'integration';

describe.skipIf(!isIntegrationTest)('Product Agent E2E', () => {
  beforeAll(async () => {
    // Verify infrastructure is running
    const healthChecks = await Promise.all([
      fetch('http://localhost:3005/health').catch(() => null),
      fetch('http://localhost:3001/health').catch(() => null),
      fetch('http://localhost:3003/health').catch(() => null),
    ]);

    const allHealthy = healthChecks.every(r => r?.ok);
    if (!allHealthy) {
      throw new Error('Infrastructure not running. Start with: docker compose up');
    }
  });

  describe('Health Checks', () => {
    it('product-agent responds on port 3005', async () => {
      const response = await fetch('http://localhost:3005/health');
      expect(response.ok).toBe(true);

      const body = await response.json();
      expect(body.status).toBe('ok');
      expect(body.service).toBe('product-agent');
    });
  });

  describe('Event Handling', () => {
    it('accepts normalized Slack events', async () => {
      const event = {
        id: 'evt_test_123',
        type: 'slack.message.app_mention',
        source: 'slack',
        timestamp: new Date().toISOString(),
        correlationId: 'test_corr_123',
        payload: {
          channelId: 'C12345', // Not in allowlist - should be ignored
          threadTs: '1234567890.123456',
          userId: 'U12345',
          text: 'Add a feature to export data',
          isThreadReply: false,
        },
      };

      const response = await fetch('http://localhost:3005/events', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Correlation-ID': event.correlationId,
          'X-Event-ID': event.id,
        },
        body: JSON.stringify(event),
      });

      expect(response.ok).toBe(true);

      const body = await response.json();
      expect(body.received).toBe(true);
      // Channel not in allowlist, so ignored
      expect(body.ignored).toBe(true);
    });

    it('rejects invalid event payloads', async () => {
      const response = await fetch('http://localhost:3005/events', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ invalid: 'payload' }),
      });

      expect(response.status).toBe(400);
    });
  });

  // Full E2E test requires:
  // 1. Valid Slack credentials
  // 2. Valid Linear credentials

# Product Agent E2E Verification

## Prerequisites

1. Docker Compose running: `docker compose up`
2. Slack workspace configured with bot token
3. Linear workspace configured with API token
4. Product-agent allowed channel configured in PRODUCT_AGENT_ALLOWED_CHANNELS

## Verification Steps

### 1. Start Infrastructure

```bash
docker compose up -d
docker compose logs -f product-agent
```

### 2. Verify Health

```bash
curl http://localhost:3005/health
# Expected: {"status":"ok","service":"product-agent"}
```

### 3. Test Conversation Flow

In the allowed Slack channel:

1. **Initial Message**: @mention the bot with a feature request
   - Example: "@product-agent Add ability to export dashboard as PDF"

2. **Verify Classification**: Bot should respond with clarifying question
   - Example: "Who needs this feature and why?"

3. **Provide Details**: Reply in thread with more context
   - Example: "Sales team needs to share dashboards with clients who don't have app access"

4. **Verify Confirmation**: Bot should show issue preview
   - Check: Title, description, acceptance criteria shown
   - Check: "Reply confirm to create" instruction

5. **Confirm**: Reply "confirm" in thread

6. **Verify Issue Created**:
   - Check Slack: Bot posts link to created issue
   - Check Linear: Issue exists with correct content
   - Check Linear: Issue has "agent-ready" label
   - Check Linear: Issue description links to Slack thread

### 4. Test Edge Cases

- **Non-allowed channel**: @mention in different channel -> polite decline
- **Question message**: "@product-agent how do I export?" -> polite decline
- **Cancellation**: Reply "nevermind" mid-conversation -> graceful exit
- **Timeout**: Don't reply for 24h -> reminder sent (hard to test manually)

## Success Criteria

- [ ] Bot responds to @mention in allowed channel
- [ ] Clarifying questions are relevant
- [ ] Confirmation shows issue preview
- [ ] Linear issue created with correct content
- [ ] Issue has agent-ready label
- [ ] Issue description includes Slack link
- [ ] Slack notification includes Linear URL
- [ ] Non-allowed channels get polite decline
