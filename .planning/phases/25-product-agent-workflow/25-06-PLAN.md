   ```typescript
   /**
    * Product Agent Event Handler
    *
    * Handles incoming Slack events from dispatcher.
    * Starts or resumes Temporal workflow based on event type.
    */

   import { NormalizedEventSchema, type PinoLogger } from '@aesir/common';
   import type { WorkflowClient } from '@temporalio/client';
   import { userReplySignal, cancelConversationSignal } from '../../temporal/signals.js';
   import { productAgentConversationWorkflow } from '../../temporal/workflows/product-agent-workflow.js';

   export interface ProductAgentEventsHandlerDeps {
     logger: PinoLogger;
     workflowClient: WorkflowClient;
     allowedChannels: string[]; // Channel allowlist from config
   }

   export interface EventsRequest {
     headers: Record<string, string | undefined>;
     rawBody: string;
   }

   export interface EventsResponse {
     status: (code: number) => { json: (body: unknown) => void };
   }

   export function createProductAgentEventsHandler(deps: ProductAgentEventsHandlerDeps) {
     const { logger, workflowClient, allowedChannels } = deps;

     return async (req: EventsRequest, res: EventsResponse): Promise<void> => {
       const correlationId = req.headers['x-correlation-id'];
       const eventId = req.headers['x-event-id'];

       const childLogger = logger.child({ endpoint: '/events', correlationId, eventId });

       try {
         const body = JSON.parse(req.rawBody);
         const parseResult = NormalizedEventSchema.safeParse(body);

         if (!parseResult.success) {
           childLogger.warn({ errors: parseResult.error.errors }, 'Invalid event');
           res.status(400).json({ error: 'Invalid event payload' });
           return;
         }

         const event = parseResult.data;

         // Only handle Slack events
         if (event.source !== 'slack') {
           childLogger.debug({ source: event.source }, 'Ignoring non-Slack event');
           res.status(200).json({ received: true, ignored: true });
           return;
         }

         // Extract Slack-specific payload
         const payload = event.payload as {
           channelId: string;
           threadTs: string;
           userId: string;
           text: string;
           isThreadReply: boolean;
         };

         // Check channel allowlist
         if (!allowedChannels.includes(payload.channelId)) {
           childLogger.info({ channelId: payload.channelId }, 'Channel not in allowlist');
           // TODO: Send polite decline message via MCP
           res.status(200).json({ received: true, ignored: true, reason: 'channel_not_allowed' });
           return;
         }

         // Determine if this is new conversation or reply
         const workflowId = `product-agent-${payload.threadTs}`;

         if (!payload.isThreadReply) {
           // New conversation - start workflow
           childLogger.info({ workflowId }, 'Starting new conversation workflow');

           await workflowClient.start(productAgentConversationWorkflow, {
             taskQueue: 'product-agent',
             workflowId,
             args: [{
               threadTs: payload.threadTs,
               channelId: payload.channelId,
               initialMessage: payload.text,
               userId: payload.userId,
               slackTeamId: event.payload.teamId || 'default',
             }],
           });
         } else {
           // Thread reply - signal existing workflow
           childLogger.info({ workflowId }, 'Signaling existing workflow');

           // Check for cancellation keywords
           const cancelKeywords = ['nevermind', 'never mind', 'cancel', 'forget it'];
           const isCancel = cancelKeywords.some(kw =>
             payload.text.toLowerCase().includes(kw)
           );

           const handle = workflowClient.getHandle(workflowId);

           if (isCancel) {
             await handle.signal(cancelConversationSignal);
           } else {
             await handle.signal(userReplySignal, payload.text);
           }
         }

         res.status(200).json({ received: true, eventId: event.id });
       } catch (error) {
         childLogger.error({ err: error }, 'Error processing event');

         // If workflow not found, that's okay - conversation may have ended
         if ((error as Error).message?.includes('not found')) {
           res.status(200).json({ received: true, workflowEnded: true });
           return;
         }

         res.status(500).json({ error: 'Internal server error' });
       }
     };
   }
   ```

   ```typescript
   import express from 'express';
   import { createPinoLogger } from '@aesir/common';
   import { Connection, Client } from '@temporalio/client';
   import { createProductAgentEventsHandler } from './api/events.js';
   import { createProductAgentWorker } from './worker.js';

   const logger = createPinoLogger({ component: 'product-agent' });
   const port = Number(process.env.PORT) || 3005;
   const allowedChannels = (process.env.PRODUCT_AGENT_ALLOWED_CHANNELS || '').split(',').filter(Boolean);

   async function main() {
     logger.info({ port, allowedChannels }, 'Starting product-agent');

     // Connect to Temporal
     const connection = await Connection.connect({
       address: process.env.TEMPORAL_ADDRESS || 'localhost:7233',
     });
     const workflowClient = new Client({ connection });

     // Start Temporal worker (runs in same process)
     const worker = await createProductAgentWorker(connection);
     worker.run(); // Don't await - runs in background

     // Create Express app
     const app = express();

     // Raw body for event validation
     app.use(express.raw({ type: 'application/json' }));

     // Health check
     app.get('/health', (_req, res) => {
       res.json({ status: 'ok', service: 'product-agent' });
     });

     // Events endpoint
     const eventsHandler = createProductAgentEventsHandler({
       logger,
       workflowClient,
       allowedChannels,
     });

     app.post('/events', async (req, res) => {
       await eventsHandler(
         { headers: req.headers as Record<string, string>, rawBody: req.body.toString() },
         { status: (code) => ({ json: (body) => res.status(code).json(body) }) }
       );
     });

     // Start server
     app.listen(port, () => {
       logger.info({ port }, 'Product-agent listening');
     });
   }

   main().catch((err) => {
     logger.error({ err }, 'Failed to start product-agent');
     process.exit(1);
   });
   ```

Create worker.ts for Temporal workflow execution:

```typescript
import { Worker, NativeConnection } from '@temporalio/worker';
import { createPinoLogger } from '@aesir/common';
import { createProductAgentCheckpointer } from './checkpointer.js';

// Import activities
import * as productAgentActivities from '../temporal/activities/product-agent-activity.js';
import * as slackActivities from '../temporal/activities/slack-activities.js';

const logger = createPinoLogger({ component: 'product-agent:worker' });

/**
 * Create Temporal worker for product-agent task queue.
 *
 * Worker executes productAgentConversationWorkflow and associated activities.
 */
export async function createProductAgentWorker(
  connection: NativeConnection
): Promise<Worker> {
  logger.info('Creating product-agent Temporal worker');

  // Initialize checkpointer before worker starts
  // This ensures it's ready when activities need it
  await createProductAgentCheckpointer();
  logger.info('Checkpointer initialized');

  const worker = await Worker.create({
    connection,
    namespace: process.env.TEMPORAL_NAMESPACE || 'default',
    taskQueue: 'product-agent',
    workflowsPath: require.resolve('../temporal/workflows/product-agent-workflow.js'),
    activities: {
      ...productAgentActivities,
      ...slackActivities,
    },
  });

  logger.info('Product-agent worker created, polling task queue: product-agent');

  return worker;
}
```

Key points:
- Polls 'product-agent' task queue
- Registers productAgentConversationWorkflow
- Registers all required activities
- Initializes checkpointer at startup (before any activity runs)
- Uses workflowsPath for workflow bundling
